173:     /**
174:      * Identifies the caller whose requests share a throttling budget.
175:      *
176:      * Evaluates user_identificator_getter on each request, using its nonempty string or integer
177:      * result. Null or an empty string falls back to REMOTE_ADDR. Identity sources are prefixed
178:      * so a user identifier cannot collide with an IP budget. The IP is normalized; forwarded
179:      * headers are ignored. This hook runs before body parsing and preprocessing and can be overridden.
180:      *
181:      * @return string
182:      * Returns a user:-prefixed identifier, an ip:-prefixed address, or ip:unknown if no valid IP is available.
183:      * Integer zero and the string "0" are valid user identifiers and share the same budget.
184:      *
185:      * @throws \UnexpectedValueException
186:      * If the getter returns a value other than string, integer or null.
187:      *
188:      * @throws \Throwable
189:      * If the getter throws an exception or error; it is reported by handleRequest() as a system failure.
190:      *
191:      * @author Oleg Schildt
192:      */
193:     protected function getRequestThrottlingKey(): string
194:     {
195:         if ($this->user_identificator_getter !== null) {
196:             $identifier = ($this->user_identificator_getter)();
197: 
198:             if (!is_string($identifier) && !is_int($identifier) && $identifier !== null) {
199:                 throw new \UnexpectedValueException("The user identifier getter must return a string, integer or null.");
200:             }

202:             if ($identifier !== null && $identifier !== "") {
203:                 return "user:" . (string) $identifier;
204:             }
205:         }

292:     protected function processInputData(): void
293:     {
294:         $uri = $_SERVER['REQUEST_URI'] ?? "";
295:         $script = $_SERVER['SCRIPT_NAME'] ?? "";
296:         $parts = is_string($uri) ? parse_url($uri) : false;
297: 
298:         if ($parts === false) {
299:             throw new SmartException("The request URI is invalid!", SmartException::ERR_CODE_INVALID_REQUEST_DATA);
300:         }
301: 
302:         $path = $parts['path'] ?? "";
303:         $query = isset($parts['query']) ? "?" . $parts['query'] : "";

306: 
307:         if (str_starts_with($path, $base)) {
308:             $path = substr($path, strlen($base));
309:         } else {
310:             $path = ltrim($path, "/");
311:         }
312: 
313:         $script_name = basename($script);
314: 
315:         if (
316:             $script_name !== ""
317:             && ($path === $script_name || str_starts_with($path, $script_name . "/"))
318:         ) {

320:         }

362:     /**
363:      * Validates complete media type syntax and an optional required charset.
364:      *
365:      * @param string $expected_type
366:      * The required type/subtype, compared without regard to case.
367:      *
368:      * @param string|null $charset
369:      * The permitted charset when explicitly supplied; null leaves encoding selection to the parser.
370:      * A missing charset is accepted in either case.
371:      *
372:      * @return void
373:      * Accepts a matching media type with syntactically valid token or quoted parameters.
374:      *
375:      * @throws SmartException
376:      * If Content-Type is missing, malformed, has another media type or an unsupported charset.
377:      *
378:      * @author Oleg Schildt
379:      */
380:     protected function validateContentType(string $expected_type, ?string $charset = null): void
381:     {
382:         $content_type = get_header("Content-Type");
383:         $token = "[!#$%&'*+.^_\\x60|~0-9A-Za-z-]+";
384:         $quoted = '"(?:[^"\\\\\r\n]|\\\\[^\r\n])*"';
385:         $parameter = '[ \t]*;[ \t]*(' . $token . ')[ \t]*=[ \t]*(' . $token . '|' . $quoted . ')';
386:         $pattern = '@^[ \t]*(' . $token . '/' . $token . ')((?:' . $parameter . ')*)[ \t]*$@D';
387: 
388:         if (!preg_match($pattern, $content_type, $matches) || strcasecmp($matches[1], $expected_type) !== 0) {
389:             throw new SmartException(
390:                 sprintf("Content type '%s' is expected!", $expected_type),
391:                 SmartException::ERR_CODE_INVALID_CONTENT_TYPE
392:             );
393:         }
394: 
395:         if ($charset === null) {
396:             return;
397:         }
398: 
399:         preg_match_all('@' . $parameter . '@', $matches[2], $parameters, PREG_SET_ORDER);
400: 
401:         foreach ($parameters as $parameter_match) {
402:             if (strcasecmp($parameter_match[1], "charset") !== 0) {
403:                 continue;
404:             }
405: 
406:             $value = $parameter_match[2];
407: 
408:             if (str_starts_with($value, '"')) {
409:                 $value = preg_replace('@\\\\(.)@s', '$1', substr($value, 1, -1));
410:             }
411: 
412:             if (strcasecmp($value, $charset) !== 0) {
413:                 throw new SmartException(
414:                     sprintf("The request charset must be '%s'!", $charset),
415:                     SmartException::ERR_CODE_INVALID_CONTENT_TYPE
416:                 );
417:             }
418:         }
419:     } // validateContentType

597:      *

643:     /**
644:      * Handles one request using public methods as actions, without a separate action registry.
645:      *
646:      * Resets request state, enforces optional throttling, parses input and invokes an action.
647:      * Canonical method names exclude lifecycle entry points and dependency accessors from dispatch.
648:      * Throttled requests receive 429 and Retry-After before body parsing or application hooks.
649:      * Structured exceptions become client errors; unexpected failures are logged.
650:      * Cleanup runs after preprocessing starts, even on failure. Response assembly failures
651:      * produce a minimal response in the handler's format with HTTP status 500.
652:      * Errors added directly to the message manager select status 400 unless an action,
653:      * a hook or exception reporting has already selected an error status.
654:      *
655:      * @return void
