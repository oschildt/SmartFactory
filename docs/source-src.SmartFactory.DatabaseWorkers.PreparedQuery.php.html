82:     /**
83:      * Compiles application bind markers into native SQL without changing quoted SQL content.
84:      *
85:      * Recognizes quoted strings/identifiers, comments, PostgreSQL dollar quotes and JSON operators,
86:      * and Oracle q/nq quoting. PostgreSQL reuses the same native index for repeated names so all uses
87:      * participate in native type inference. Parameter values are never inserted into SQL text.
88:      *
89:      * @param string $sql
90:      * The application SQL containing either ? or :name markers.
91:      *
92:      * @param string $dialect
93:      * The SQL dialect: mysql, mssql, postgres or oracle.
94:      *
95:      * @param bool $backslash_escapes
96:      * Whether MySQL backslash escapes are active. Other dialects apply their own quoting rules.
97:      *
98:      * @return self
99:      * Returns the original SQL, translated SQL and marker occurrence mapping.
100:      *
101:      * @throws DBWorkerException
102:      * If marker styles are mixed, native numbered markers are used, a quoted construct or comment is
103:      * unterminated, or placeholders occur inside an executable MySQL comment.
104:      *
105:      * @author Oleg Schildt
106:      */
107:     public static function compile(string $sql, string $dialect, bool $backslash_escapes = true): self
108:     {
109:         // Scan SQL constructs before recognizing markers so quoted text and comments remain untouched.
110:         $output = '';
111:         $names = [];
112:         $named = false;
113:         $positional = false;
114:         $named_indexes = [];
115:         $operand = false;
116:         $length = strlen($sql);
117: 
118:         // Track expression boundaries so PostgreSQL JSON operators are not mistaken for bind markers.
119:         $keywords = array_fill_keys(
120:             explode(
121:                 ' ',
122:                 'SELECT WHERE AND OR NOT WHEN THEN ELSE ON BY AS HAVING LIMIT OFFSET VALUES SET RETURNING DISTINCT INTO UNION ALL CASE EXISTS BETWEEN LIKE ILIKE IS IN UPDATE INSERT DELETE FROM JOIN CROSS LEFT RIGHT INNER OUTER FULL USING TOP FETCH NEXT ROWS ONLY ASC DESC'
123:             ),
124:             true
125:         );
126: 
127:         for ($i = 0; $i < $length;) {
128:             $start = $i;
129:             $char = $sql[$i];
130:             $next = $sql[$i + 1] ?? '';
131: 
132:             if (
133:                 (
134:                     $char === '-'
135:                     && $next === '-'
136:                     && ($dialect !== 'mysql' || $i + 2 === $length || ctype_space($sql[$i + 2]))
137:                 )
138:                 || ($dialect === 'mysql' && $char === '#')
139:             ) {
140:                 // Skip a line comment as opaque SQL, including any marker-like text in it.
141:                 $end = strpos($sql, "\n", $i);
142:                 $i = $end === false ? $length : $end;
143:             } elseif ($char === '/' && $next === '*') {
144:                 $i += 2;
145: 
146:                 // Only dialects with nested block comments increase the nesting depth.
147:                 $depth = 1;
148: 
149:                 while ($i < $length && $depth > 0) {
150:                     if (substr($sql, $i, 2) === '/*' && in_array($dialect, ['postgres', 'mssql'], true)) {
151:                         $depth++;
152:                         $i += 2;
153:                     } elseif (substr($sql, $i, 2) === '*/') {
154:                         $depth--;
155:                         $i += 2;
156:                     } else {
157:                         $i++;
158:                     }
159:                 }
160: 
161:                 if ($depth !== 0) {
162:                     self::invalid('Unterminated SQL comment.');
163:                 }
164: 
165:                 if (
166:                     $dialect === 'mysql'
167:                     && substr($sql, $start, 3) === '/*!'
168:                     && preg_match('/\?|:[a-zA-Z_]/', substr($sql, $start, $i - $start))
169:                 ) {
170:                     self::invalid('Placeholders inside executable MySQL comments are not supported.');
171:                 }
172:             } elseif (
173:                 $dialect === 'oracle'
174:                 && preg_match("/\\A(?:nq|q)'(.)/is", substr($sql, $i), $match)
175:                 && ($i === 0 || !preg_match('/[a-zA-Z0-9_]/', $sql[$i - 1]))
176:             ) {
177:                 // Oracle alternative quotes use either a paired delimiter or a custom closing character.
178:                 $open = $match[1];
179:                 $close = ['[' => ']', '(' => ')', '{' => '}', '<' => '>'][$open] ?? $open;
180:                 $end = strpos($sql, $close . "'", $i + strlen($match[0]));
181: 
182:                 if ($end === false) {
183:                     self::invalid('Unterminated Oracle quoted string.');
184:                 }
185: 
186:                 $i = $end + 2;
187:                 $operand = true;
188:             } elseif (
189:                 $char === "'"
190:                 || $char === '"'
191:                 || ($dialect === 'mysql' && $char === '`')
192:                 || ($dialect === 'mssql' && $char === '[')
193:             ) {
194:                 // Handle doubled delimiters and only the backslash escapes enabled for this dialect.
195:                 $close = $char === '[' ? ']' : $char;
196:                 $escape = ($dialect === 'mysql' && $backslash_escapes)
197:                     || (
198:                         $dialect === 'postgres'
199:                         && $char === "'"
200:                         && $i > 0
201:                         && strtolower($sql[$i - 1]) === 'e'
202:                         && ($i < 2 || !preg_match('/[a-zA-Z0-9_]/', $sql[$i - 2]))
203:                     );
204:                 $i++;
205:                 $closed = false;
206: 
207:                 while ($i < $length) {
208:                     if ($escape && $sql[$i] === '\\') {
209:                         $i += 2;
210:                     } elseif ($sql[$i] === $close) {
211:                         if (($sql[$i + 1] ?? '') === $close) {
212:                             $i += 2;
213:                         } else {
214:                             $i++;
215:                             $closed = true;
216: 
217:                             break;
218:                         }
219:                     } else {
220:                         $i++;
221:                     }
222:                 }
223: 
224:                 if (!$closed) {
225:                     self::invalid('Unterminated SQL string or identifier.');
226:                 }
227: 
228:                 $operand = true;
229:             } elseif (
230:                 $dialect === 'postgres'
231:                 && $char === '$'
232:                 && preg_match('/\A\$(?:[a-zA-Z_][a-zA-Z0-9_]*)?\$/', substr($sql, $i), $match)
233:             ) {
234:                 // Keep the entire PostgreSQL dollar-quoted string unchanged.
235:                 $end = strpos($sql, $match[0], $i + strlen($match[0]));
236: 
237:                 if ($end === false) {
238:                     self::invalid('Unterminated PostgreSQL dollar-quoted string.');
239:                 }
240: 
241:                 $i = $end + strlen($match[0]);
242:                 $operand = true;
243:             } elseif ($char === '$' && ctype_digit($next)) {
244:                 self::invalid('Use ? for positional placeholders; native $1 placeholders are not accepted.');
245:             } elseif ($char === ':' && ($next === ':' || $next === '=')) {
246:                 $i += 2; // PostgreSQL casts and PL/SQL assignments.
247:             } elseif ($char === ':' && ctype_digit($next)) {
248:                 self::invalid('Use ? or :name for placeholders.');
249:             } elseif ($char === ':' && preg_match('/\A:([a-zA-Z_][a-zA-Z0-9_]*)/', substr($sql, $i), $match)) {
250:                 // Record each named occurrence, but reuse the native PostgreSQL index for repeated names.
251:                 $named = true;
252:                 $names[] = $match[1];
253:                 $named_indexes[$match[1]] ??= count($named_indexes) + 1;
254:                 $output .= self::marker($dialect, $dialect === 'postgres' ? $named_indexes[$match[1]] : count($names));
255:                 $i += strlen($match[0]);
256:                 $operand = true;
257: 
258:                 continue;
259:             } elseif ($char === '?') {
260:                 // PostgreSQL JSON operators are SQL, not bind markers. An operand on the
261:                 // left distinguishes the bare ? operator from a positional placeholder.
262:                 if (
263:                     $dialect === 'postgres'
264:                     && ($operand || in_array($next, ['|', '&'], true) || ($sql[$i - 1] ?? '') === '@')
265:                 ) {
266:                     $i += in_array($next, ['|', '&'], true) ? 2 : 1;
267:                     $operand = false;
268:                 } else {
269:                     // Only a real positional marker contributes another bound value.
270:                     $positional = true;
271:                     $names[] = null;
272:                     $output .= self::marker($dialect, count($names));
273:                     $i++;
274:                     $operand = true;
275: 
276:                     continue;
277:                 }
278:             } elseif (preg_match('/[a-zA-Z_]/', $char)) {
279:                 preg_match('/\A[a-zA-Z_][a-zA-Z0-9_$#]*/', substr($sql, $i), $match);
280: 
281:                 $i += strlen($match[0]);
282: 
283:                 // A keyword requiring an expression cannot be the left operand of a JSON operator.
284:                 $operand = !isset($keywords[strtoupper($match[0])]);
285:             } else {
286:                 if (!ctype_space($char)) {
287:                     $operand = ctype_digit($char) || $char === ')' || $char === ']';
288:                 }
289: 
290:                 $i++;
291:             }
292: 
293:             // Preserve the exact bytes of SQL regions that do not need marker translation.
294:             $output .= substr($sql, $start, $i - $start);
295:         }
296: 
297:         // Reject a mixed public syntax after scanning all quoted/commented regions.
298:         if ($named && $positional) {
299:             self::invalid('Named and positional placeholders cannot be mixed.');
300:         }
301: 
302:         return new self($sql, $output, $names, $named);
303:     } // compile

305:     /**
306:      * Formats a native bind marker for the selected database dialect.
307:      *
308:      * @param string $dialect
309:      * The native SQL dialect.
310:      *
311:      * @param int $index
312:      * The one-based native parameter index.
313:      *
314:      * @return string
315:      * Returns ?, $n or :pn according to the driver.
316:      *
317:      * @author Oleg Schildt
318:      */
319:     private static function marker(string $dialect, int $index): string
320:     {
321:         // Translate only the placeholder syntax; argument values never become part of SQL text.
322:         return match ($dialect) {
323:             'postgres' => '$' . $index,
324:             'oracle' => ':p' . $index,
325:             default => '?'
326:         };
327:     } // marker

329:     /**
330:      * Validates arguments and expands repeated named occurrences into ordered values.
331:      *
332:      * A supplied NULL is recognized as a value. Named array keys may start with one colon.
333:      * Duplicate normalized names are rejected, and floating-point values must be finite.
334:      *
335:      * @param array<array-key, mixed> $args
336:      * The variadic argument array received by execute_prepared_query(), optionally containing a single argument array.
337:      *
338:      * @return list<scalar|null|BlobValue|ClobValue>
339:      * Returns values in SQL occurrence order; a repeated name contributes the same value for each occurrence.
340:      *
341:      * @throws DBWorkerException
342:      * If argument names or counts do not match the placeholders, argument styles are mixed, positional
343:      * keys are not contiguous, names are invalid or duplicated, or values have unsupported types or non-finite floats.
344:      *
345:      * @author Oleg Schildt
346:      */
347:     public function arguments(array $args): array
348:     {
349:         // Normalize the supported call forms before checking the exact set of required parameters.
350:         if (array_is_list($args) && count($args) === 1 && is_array($args[0])) {
351:             $args = $args[0];
352:         }
353: 
354:         if (!$this->named) {
355:             if (!array_is_list($args) || count($args) !== count($this->names)) {
356:                 self::invalid('Positional parameters must be a contiguous list matching the number of placeholders.');
357:             }
358: 
359:             $values = $args;
360:         } else {
361:             $provided = [];
362: 
363:             foreach ($args as $name => $value) {
364:                 if (!is_string($name)) {
365:                     self::invalid('Named placeholders require named arguments or an associative array.');
366:                 }
367: 
368:                 $name = str_starts_with($name, ':') ? substr($name, 1) : $name;
369: 
370:                 if (!preg_match('/\A[a-zA-Z_][a-zA-Z0-9_]*\z/', $name) || array_key_exists($name, $provided)) {
371:                     self::invalid('Invalid or duplicate parameter name.');
372:                 }
373: 
374:                 $provided[$name] = $value;
375:             }
376: 
377:             // Validate unique names before expanding repeated SQL occurrences.
378:             $required = array_fill_keys($this->names, true);
379: 
380:             if (array_diff_key($required, $provided) || array_diff_key($provided, $required)) {
381:                 self::invalid('Supplied parameter names must exactly match the query placeholders.');
382:             }
383: 
384:             $values = array_map(static fn (string $name): mixed => $provided[$name], $this->names);
385:         }
386: 
387:         // Reject unsupported objects and non-finite floats before any native execution.
388:         foreach ($values as $value) {
389:             if ($value !== null && !is_scalar($value) && !$value instanceof ClobValue && !$value instanceof BlobValue) {
390:                 self::invalid('Unsupported parameter value; wrap streams in ClobValue or BlobValue.');
391:             }
392: 
393:             if (is_float($value) && !is_finite($value)) {
394:                 self::invalid('Numeric parameters must be finite.');
395:             }
396:         }
397: 
398:         return $values;
399:     } // arguments
