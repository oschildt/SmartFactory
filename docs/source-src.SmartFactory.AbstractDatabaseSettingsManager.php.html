56:      * Validates an unquoted SQL table or column name before composing a query.

82:     /**
83:      * Obtains a connected clone without replacing the caller's statements or results.
84:      *
85:      * @return DBWorker
86:      * Returns a worker sharing the connection and its transaction state.
87:      *
88:      * @throws \LogicException

95:      */
96:     protected function databaseWorker(): DBWorker
97:     {
98:         if ($this->dbworker === null) {
99:             throw new \LogicException("The 'dbworker' is not specified.");
100:         }
101: 
102:         if (!$this->dbworker->is_connected()) {
103:             $this->dbworker->connect();
104:         }

106:         return $this->dbworker->create_clone();

109:     /**
110:      * Executes one aggregate save inside an owned or caller-managed transaction.
111:      *
112:      * @param callable(DBWorker): array<string, mixed> $action
113:      * The action which locks, merges and saves the complete settings aggregate.
114:      *
115:      * @return array<string, mixed>
116:      * Returns the saved snapshot after committing an owned transaction.
117:      *
118:      * @throws \LogicException
119:      * If a required external transaction is absent or an external transaction does
120:      * not use READ COMMITTED, which is required for fresh reads after acquiring a lock.
121:      *
122:      * @throws \Throwable
123:      * If connecting, saving or committing fails. Owned transactions are rolled back;
124:      * callers retain rollback responsibility for external transactions after any failure.
125:      *
126:      * @author Oleg Schildt
127:      */
128:     protected function withTransaction(callable $action): array
129:     {
130:         $worker = $this->databaseWorker();
131:         $level = $worker->get_transaction_isolation_level();
132:         $own_transaction = $level === null;
133: 
134:         if (
135:             ($this->no_own_transaction && $own_transaction)
136:             || ($level !== null && $level !== DBWorker::ISOLATION_READ_COMMITTED)
137:         ) {
138:             throw new \LogicException("Settings require a READ COMMITTED transaction; the configured caller transaction is absent or incompatible.");
139:         }
140: 
141:         if ($own_transaction) {
142:             $worker->start_transaction(DBWorker::ISOLATION_READ_COMMITTED);
143:         }
144: 
145:         try {
146:             $data = $action($worker);
147: 
148:             if ($own_transaction) {
149:                 $worker->commit_transaction();
150:             }
151: 
152:             return $data;
153:         } catch (\Throwable $exception) {
154:             if ($own_transaction && $worker->get_transaction_isolation_level() !== null) {
155:                 try {
156:                     $worker->rollback_transaction();
157:                 } catch (\Throwable) {
158:                     // Retain the original failure if connection recovery is also unsuccessful.
159:                 }
160:             }
161: 
162:             throw $exception;
163:         } finally {
164:             $worker->free_prepared_query();
165:         }
166:     } // withTransaction
