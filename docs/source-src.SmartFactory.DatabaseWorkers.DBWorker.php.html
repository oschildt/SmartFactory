674:     /**
675:      * Initializes the DBWorker with connection parameters and driver-specific options.
676:      *
677:      * @param array $parameters
678:      * Connection settings and options as an associative array.
679:      *
680:      * - $parameters["db_server"] - server address or Oracle connection descriptor.
681:      * - $parameters["db_port"] - optional TCP port.
682:      * - $parameters["db_name"] - database name or Oracle service name.
683:      * - $parameters["db_user"] - database user name.
684:      * - $parameters["db_password"] - database password; an empty password is permitted.
685:      * - $parameters["db_schema"] - optional schema name.
686:      * - $parameters["read_only"] - requests a supported native read-only mode.
687:      * - $parameters["default_isolation_level"] - default for start_transaction(); READ COMMITTED when not configured.
688:      * - $parameters["query_logger"] - optional query timing callback.
689:      *
690:      * Timeout and isolation options depend on the driver. Supplied false and empty values
691:      * replace previous settings. This method does not open or replace a connection.
692:      *
693:      * @return void
694:      * Updates the supplied connection settings and the default transaction isolation level.
695:      *
696:      * @throws DBWorkerException
697:      * If the port is not a number between 1 and 65535 or the default isolation is not a supported string value.
698:      *
699:      * @see DBWorker::connect()
700:      *
701:      * @author Oleg Schildt
702:      */
703:     public function init(array $parameters): void
704:     {
705:         // Apply explicitly supplied values so empty strings and false do not silently preserve old settings.
706:         foreach (['db_server', 'db_port', 'db_name', 'db_user', 'db_password'] as $key) {
707:             if (array_key_exists($key, $parameters)) {
708:                 $this->$key = (string) $parameters[$key];
709:             }
710:         }
711: 
712:         if (
713:             $this->db_port !== ''
714:             && (!ctype_digit($this->db_port) || (int) $this->db_port < 1 || (int) $this->db_port > 65535)
715:         ) {
716:             throw new DBWorkerException('Database port must be between 1 and 65535.', self::ERR_WRONG_DATA_FORMAT);
717:         }
718: 
719:         if (array_key_exists('read_only', $parameters)) {
720:             $this->read_only = (bool) $parameters['read_only'];
721:         }
722: 
723:         if (array_key_exists('default_isolation_level', $parameters)) {
724:             $level = $parameters['default_isolation_level'];
725: 
726:             if (!is_string($level)) {
727:                 throw new DBWorkerException('The default isolation level must be a string.', self::ERR_WRONG_DATA_FORMAT);
728:             }
729: 
730:             $level = strtoupper(trim($level));
731: 
732:             if (!in_array($level, [self::ISOLATION_READ_COMMITTED, self::ISOLATION_REPEATABLE_READ,
733:                 self::ISOLATION_SERIALIZABLE], true)) {
734:                 throw new DBWorkerException('Unsupported default transaction isolation level: ' . $level, self::ERR_WRONG_DATA_FORMAT);
735:             }
736: 
737:             $this->default_isolation_level = $level;
738:         }
739: 
740:         // Retain driver options alongside the common connection settings.
741:         $this->options = array_replace($this->options, $parameters);
742:     } // init

1030:     /**
1031:      * Executes the prepared SQL query with validated bind values.
1032:      *
1033:      * Example:
1034:      * ```php
1035:      * $dbw->prepare_query("SELECT id FROM users WHERE first_name = :name OR last_name = :name");
1036:      * $dbw->execute_prepared_query(name: "John");
1037:      * $dbw->free_result();
1038:      * $dbw->execute_prepared_query(["name" => "Smith"]);
1039:      * ```
1040:      *
1041:      * Missing/extra arguments and mixed styles cause an exception before execution.
1042:      * Caller-owned streams remain open; seekable input positions are preserved.
1043:      *
1044:      * @param mixed ...$args
1045:      * Positional arguments, a contiguous positional array, PHP named arguments, or one associative array.
1046:      * The argument style must match the prepared query. Names must match exactly; their order is not important.
1047:      * Explicit NULL values are accepted. Repeated named markers use the same supplied value.
1048:      * Strings and scalar values are supported; wrap binary/text LOB values or streams in BlobValue/ClobValue.
1049:      *
1050:      * @return void
1051:      *
1052:      * @throws DBWorkerException
1053:      * It might throw an exception if the connection, query or parameter validation fails.
1054:      *
1055:      * @see DBWorker::prepare_query()
1056:      * @see DBWorker::free_prepared_query()
1057:      * @see BlobValue
1058:      * @see ClobValue
1059:      *
1060:      * @author Oleg Schildt
1061:      */
1062:     public function execute_prepared_query(...$args): void
1063:     {
1064:         // Validate all arguments before releasing the previous result or executing any SQL.
1065:         $this->check_connection();
1066: 
1067:         if ($this->compiled_query === null) {
1068:             throw $this->query_error('No prepared query defined.');
1069:         }
1070: 
1071:         $values = $this->compiled_query->arguments($args);
1072: 
1073:         $this->free_result();
1074: 
1075:         $this->last_query = $this->compiled_query->original;
1076:         $this->last_insert_id = null;
1077:         $temporary = [];
1078: 
1079:         try {
1080:             // Repeated named streams must also work for non-seekable input.
1081:             // Reuse a single spool for every occurrence of the same non-seekable source.
1082:             $spooled = [];
1083: 
1084:             foreach ($values as &$value) {
1085:                 if (
1086:                     ($value instanceof BlobValue || $value instanceof ClobValue) && is_resource($value->value)
1087:                     && $this->stream_position($value->value) === null
1088:                 ) {
1089:                     $id = get_resource_id($value->value);
1090: 
1091:                     if (!isset($spooled[$id])) {
1092:                         $target = tmpfile();
1093: 
1094:                         if ($target === false) {
1095:                             throw $this->stream_error('Cannot create a temporary LOB stream.');
1096:                         }
1097: 
1098:                         $temporary[] = $target;
1099: 
1100:                         $this->write_chunks($value->value, fn (string $chunk) => $this->write_all($target, $chunk));
1101: 
1102:                         rewind($target);
1103: 
1104:                         $spooled[$id] = $target;
1105:                     }
1106: 
1107:                     $value = $value instanceof BlobValue ? new BlobValue($spooled[$id]) : new ClobValue($spooled[$id]);
1108:                 }
1109:             }
1110: 
1111:             unset($value);
1112: 
1113:             $this->timed(fn () => $this->native_execute($values), $values);
1114:         } finally {
1115:             // Only internally created temporary streams are closed here.
1116:             foreach ($temporary as $stream) {
1117:                 fclose($stream);
1118:             }
1119:         }
1120:     } // execute_prepared_query
