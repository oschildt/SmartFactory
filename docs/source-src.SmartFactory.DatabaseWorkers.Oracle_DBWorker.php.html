15:  *

187:     /**
188:      * Establishes the database connection using the initialized settings.
189:      *
190:      * Uses AL32UTF8 and deterministic NLS date/time and numeric formats.
191:      * A complete connection_string is used as supplied. Otherwise db_server may contain an
192:      * Easy Connect descriptor, or a host combined with the db_name service. An optional
193:      * db_schema changes CURRENT_SCHEMA. A read-only worker starts a managed read-only transaction.
194:      * An already connected worker keeps its existing connection.
195:      *
196:      * @return void
197:      * Does not return a value.
198:      *
199:      * @throws DBWorkerException
200:      * If connection settings are incomplete, connection or session initialization fails, the requested
201:      * call timeout cannot be configured, or a read-only transaction cannot be started.
202:      *
203:      * @see DBWorker::init()
204:      * @see Oracle_DBWorker::is_connected()
205:      * @see DBWorker::close_connection()
206:      *
207:      * @author Oleg Schildt
208:      */
209:     public function connect(): void
210:     {
211:         // Reuse an attached connection and validate the settings before creating a new native handle.
212:         if ($this->is_connected()) {
213:             return;
214:         }
215: 
216:         $dsn = (string) ($this->options['connection_string'] ?? $this->db_server);
217: 
218:         if ($dsn === '') {
219:             $dsn = $this->db_name;
220:         } elseif (
221:             !isset($this->options['connection_string'])
222:             && !str_contains($dsn, '/')
223:             && !str_contains($dsn, '(')
224:             && $this->db_name !== ''
225:         ) {
226:             $dsn .= ($this->db_port === '' ? '' : ':' . (int) $this->db_port) . '/' . $this->db_name;
227:         }
228: 
229:         if ($dsn === '' || $this->db_user === '') {
230:             throw new DBWorkerException('Connection data is incomplete.', self::ERR_CONNECTION_DATA_INCOMPLETE);
231:         }
232: 
233:         $db = @oci_new_connect($this->db_user, $this->db_password, $dsn, 'AL32UTF8');
234: 
235:         if ($db === false) {
236:             throw new DBWorkerException($this->errors(), self::ERR_CONNECTION_FAILED);
237:         }
238: 
239:         // Share ownership and transaction state, rather than independent auto-commit flags in each clone.
240:         $this->connection_state = new DBConnectionState($db, static fn ($db) => oci_close($db));
241: 
242:         try {
243:             $this->control("ALTER SESSION SET NLS_DATE_FORMAT = 'YYYY-MM-DD HH24:MI:SS'");
244: 
245:             $this->control("ALTER SESSION SET NLS_TIMESTAMP_FORMAT = 'YYYY-MM-DD HH24:MI:SS'");
246: 
247:             $this->control("ALTER SESSION SET NLS_TIMESTAMP_TZ_FORMAT = 'YYYY-MM-DD HH24:MI:SS TZH:TZM'");
248: 
249:             $this->control("ALTER SESSION SET NLS_NUMERIC_CHARACTERS = '.,'");
250: 
251:             $level = strtoupper((string) ($this->options['isolation_level'] ?? 'READ COMMITTED'));
252: 
253:             if (!in_array($level, ['READ COMMITTED', 'SERIALIZABLE'], true)) {
254:                 throw new DBWorkerException('Invalid Oracle isolation level.', self::ERR_WRONG_DATA_FORMAT);
255:             }
256: 
257:             $this->control('ALTER SESSION SET ISOLATION_LEVEL = ' . $level);
258: 
259:             $this->connection_state->attributes['isolation_level'] = $level;
260: 
261:             if (isset($this->options['db_schema'])) {
262:                 $this->use_database((string) $this->options['db_schema']);
263:             }
264: 
265:             if (isset($this->options['query_timeout'])) {
266:                 if (
267:                     !function_exists('oci_set_call_timeout')
268:                     || !@oci_set_call_timeout($db, (int) $this->options['query_timeout'] * 1000)
269:                 ) {
270:                     throw $this->query_error('This Oracle client cannot configure the requested call timeout.');
271:                 }
272:             }
273: 
274:             if ($this->read_only) {
275:                 $this->start_transaction();
276:             }
277:         } catch (\Throwable $error) {
278:             $this->connection_state = null;
279: 
280:             throw $error;
281:         }
282:     } // connect

284:     /**
285:      * Executes an internal Oracle session or transaction command without replacing the public result.
286:      *
287:      * The temporary OCI8 statement is always released.
288:      *
289:      * @param string $sql
290:      * The internal SQL command.
291:      *
292:      * @param int $mode
293:      * The OCI8 execution mode. The default commits on success; transaction commands can disable automatic commit.
294:      *
295:      * @return void
296:      * Does not return a value.
297:      *
298:      * @throws DBWorkerException
299:      * If parsing or executing the command fails.
300:      *
301:      * @author Oleg Schildt
302:      */
303:     private function control(string $sql, int $mode = OCI_COMMIT_ON_SUCCESS): void
304:     {
305:         // Use a temporary native handle so an internal command cannot replace the caller's active result.
306:         $stmt = @oci_parse($this->connection_state->handle, $sql);
307: 
308:         if ($stmt === false) {
309:             throw $this->query_error($this->errors($this->connection_state->handle));
310:         }
311: 
312:         try {
313:             if (!@oci_execute($stmt, $mode)) {
314:                 throw $this->query_error($this->errors($stmt));
315:             }
316:         } finally {
317:             oci_free_statement($stmt);
318:         }
319:     } // control

359:         $this->check_connection();

373:     /**
374:      * Normalizes Oracle statement terminators while preserving PL/SQL blocks.
375:      *
376:      * @param string $sql

389:                 $this->sql_body($sql)

397:     /**
398:      * Adds RETURNING for the discovered identity column of an eligible Oracle INSERT VALUES statement.
399:      *
400:      * Respects quoted and schema-qualified table names. Discovery uses ALL_TAB_COLUMNS for
401:      * the actual owner and table. Existing RETURNING clauses and auto_returning => false
402:      * disable automatic extension; no column named id is assumed.
403:      *
404:      * @param string $sql
405:      * The normalized SQL statement to inspect.
406:      *
407:      * @return string
408:      * Returns SQL extended with an output bind, or the original SQL when automatic handling does not apply.
409:      *
410:      * @throws DBWorkerException
411:      * If identity-column metadata cannot be read.
412:      *
413:      * @author Oleg Schildt
414:      */
415:     private function with_identity(string $sql): string
416:     {
417:         // Capture an identifier from the INSERT itself instead of consulting unrelated session sequence state.
418:         $this->return_identity = false;
419:         $table = $this->insert_table($sql);
420: 
421:         if (
422:             $table === null || ($this->options['auto_returning'] ?? true) === false
423:             || preg_match('/\bRETURNING\b/i', $sql) || !preg_match('/\bVALUES\s*\(/i', $sql)
424:         ) {
425:             return $sql;
426:         }
427: 
428:         preg_match_all('/"((?:[^"]|"")+)"|([a-zA-Z_][a-zA-Z0-9_$#]*)/', $table, $matches, PREG_SET_ORDER);
429: 
430:         $parts = array_map(
431:             static fn (array $part): string => $part[1] !== ''
432:                     ? str_replace('""', '"', $part[1]) : strtoupper($part[2]),
433:             $matches
434:         );
435:         $name = array_pop($parts);
436:         $owner = $parts[0] ?? $this->get_schema();
437:         $cache = &$this->connection_state->attributes['identity_columns'];
438:         $key = $owner . '.' . $name;
439: 
440:         if (!array_key_exists($key, $cache ?? [])) {
441:             $stmt = @oci_parse(
442:                 $this->connection_state->handle,
443:                 "SELECT column_name FROM all_tab_columns WHERE owner = :owner AND table_name = :table_name AND identity_column = 'YES' ORDER BY column_id"
444:             );
445: 
446:             if ($stmt === false) {
447:                 throw $this->query_error($this->errors($this->connection_state->handle));
448:             }
449: 
450:             try {
451:                 oci_bind_by_name($stmt, ':owner', $owner);
452: 
453:                 oci_bind_by_name($stmt, ':table_name', $name);
454: 
455:                 if (!@oci_execute($stmt, OCI_NO_AUTO_COMMIT)) {
456:                     throw $this->query_error($this->errors($stmt));
457:                 }
458: 
459:                 $row = oci_fetch_row($stmt);
460:                 $cache[$key] = $row === false ? null : $row[0];
461:             } finally {
462:                 oci_free_statement($stmt);
463:             }
464:         }
465: 
466:         if ($cache[$key] !== null) {
467:             $this->return_identity = true;
468:             $sql = rtrim($sql, " \t\r\n;") . ' RETURNING "'
469:                 . str_replace('"', '""', $cache[$key]) . '" INTO :sf_generated_id';
470:         }
471: 
472:         return $sql;
473:     } // with_identity

503:      *
504:      * @return void

506:      *
507:      * @throws DBWorkerException

509:      *
510:      * @see DBWorker::execute_query()
511:      *

514:     protected function native_query(string $sql): void
515:     {
516:         // Execute the native statement, then expose its result through the shared result contract.
517:         $sql = $this->with_identity($this->normalize_sql($sql));

520:         if ($stmt === false) {

527:             if (!@oci_execute($stmt, $this->execution_mode())) {
528:                 throw $this->query_error($this->errors($stmt));
529:             }

533:             $this->accept_result($stmt);
534:         } catch (\Throwable $error) {

572:     /**
573:      * Binds validated values and executes the native prepared statement.
574:      *
575:      * Binds every OCI8 descriptor through its own persistent array entry. Explicit LOB
576:      * wrappers and plain strings longer than 4000 bytes use temporary LOB descriptors.
577:      * Input locators remain alive while the statement or result can reference them.
578:      * Execution mode is chosen from the connection-wide transaction flag.
579:      *
580:      * @param list<scalar|null|BlobValue|ClobValue> $values
581:      * The normalized values in SQL occurrence order, including any explicit LOB wrappers.
582:      *
583:      * @return void
584:      * Does not return a value.
585:      *
586:      * @throws DBWorkerException
587:      * If binding, LOB processing or native execution fails.
588:      *
589:      * @see DBWorker::execute_prepared_query()
590:      *
591:      * @author Oleg Schildt
592:      */
593:     protected function native_execute(array $values): void
594:     {
595:         // Bind stable per-position values so repeated application names cannot share a loop variable by accident.
596:         // Retire previous input locators only after the old result has been released.
597:         $this->free_lobs();
598: 
599:         $this->bound_values = [];
600: 
601:         try {
602:             foreach ($values as $index => $value) {
603:                 $name = ':p' . ($index + 1);
604: 
605:                 if (
606:                     $value instanceof ClobValue
607:                     || $value instanceof BlobValue
608:                     || (is_string($value) && strlen($value) > 4000)
609:                 ) {
610:                     $binary = $value instanceof BlobValue;
611:                     $source = is_string($value) ? $value : $value->value;
612:                     $lob = oci_new_descriptor($this->connection_state->handle, OCI_D_LOB);
613: 
614:                     if ($lob === false) {
615:                         throw $this->stream_error('Cannot allocate an Oracle LOB descriptor.');
616:                     }
617: 
618:                     $this->lob_descriptors[$index] = $lob;
619: 
620:                     if (!$lob->writeTemporary('', $binary ? OCI_TEMP_BLOB : OCI_TEMP_CLOB)) {
621:                         throw $this->stream_error('Cannot create a temporary Oracle LOB.');
622:                     }
623: 
624:                     $this->write_chunks(
625:                         $source,
626:                         function (string $chunk) use ($index, $binary): void {
627:                             if (!$binary) {
628:                                 $written = $this->lob_descriptors[$index]->write($chunk);
629: 
630:                                 // OCI reports UCS-2 units for CLOBs, including surrogate pairs,
631:                                 // whereas the PHP input string is measured in UTF-8 bytes.
632:                                 $units = strlen(mb_convert_encoding($chunk, 'UTF-16LE', 'UTF-8')) / 2;
633: 
634:                                 if ($written === false || $written !== (int) $units) {
635:                                     throw $this->stream_error('Incomplete Oracle CLOB write.');
636:                                 }
637: 
638:                                 return;
639:                             }
640: 
641:                             $offset = 0;
642: 
643:                             while ($offset < strlen($chunk)) {
644:                                 $written = $this->lob_descriptors[$index]->write(substr($chunk, $offset));
645: 
646:                                 if ($written === false || $written === 0) {
647:                                     throw $this->stream_error('Cannot write Oracle LOB.');
648:                                 }
649: 
650:                                 $offset += $written;
651:                             }
652:                         },
653:                         !$binary
654:                     );
655: 
656:                     // Bind each descriptor's own stable array location, never the loop variable.
657:                     $ok = @oci_bind_by_name(
658:                         $this->statement,
659:                         $name,
660:                         $this->lob_descriptors[$index],
661:                         -1,
662:                         $binary ? OCI_B_BLOB : OCI_B_CLOB
663:                     );
664:                 } else {

668:                         $this->statement,
669:                         $name,

671:                         -1,

727:     /**
728:      * Captures Oracle column metadata, DML counts and an automatically returned identity.
729:      *
730:      * Ordinary uppercase Oracle column names are exposed in lowercase. Case-distinct quoted
731:      * aliases remain distinct. LOB contents are not loaded while inspecting metadata.

733:      * @param resource $statement
734:      * The successfully executed OCI8 statement resource.
735:      *
736:      * @return void
737:      * Does not return a value.
738:      *
739:      * @throws DBWorkerException
740:      * If the returned identity exceeds the PHP integer range.
741:      *
742:      * @author Oleg Schildt
743:      */

745:     {
746:         // Record metadata without advancing the application row cursor.
747:         $names = [];

750:             $names[] = oci_field_name($statement, $i);

754:         $metadata = [];

762:                 true

764:             $binary = in_array($type, ['raw', 'long raw', 'blob', 'bfile'], true);

767:             // Keep distinct quoted aliases; normal Oracle identifiers use lower-case API keys.

770:                 'name' => $api_name,
771:                 'type' => $type,

773:                 'prec' => oci_field_precision($statement, $num),

775:                 'numeric' => (int) $numeric,
776:                 'binary' => (int) $binary,
777:                 'datetime' => (int) $datetime,
778:                 'string' => (int) in_array(
779:                     $type,
780:                     ['varchar2', 'nvarchar2', 'char', 'nchar', 'clob', 'nclob', 'long'],
781:                     true
782:                 )
783:             ];

1001:     /**
1002:      * Copies one native field value to the destination stream.
1003:      *
1004:      * Drivers override this hook to read native LOB objects without first materializing their contents.
1005:      *
1006:      * @param int $index
1007:      * The zero-based index of the field in the current row.
1008:      *
1009:      * @param resource $stream
1010:      * The caller-owned writable destination stream.
1011:      *
1012:      * @return int
1013:      * Returns the number of bytes written; NULL fields produce zero bytes.
1014:      *
1015:      * @throws DBWorkerException
1016:      * If a stream is invalid or a read/write operation fails.
1017:      *
1018:      * @author Oleg Schildt
1019:      */
1020:     protected function copy_value_to_stream(int $index, $stream): int
1021:     {
1022:         // Read the native locator incrementally and count bytes actually written to the destination.
1023:         $lob = $this->row[$index];
1024: 
1025:         if (!$lob instanceof \OCILob) {
1026:             return parent::copy_value_to_stream($index, $stream);
1027:         }
1028: 
1029:         if (!$lob->rewind()) {
1030:             throw $this->stream_error('Cannot rewind Oracle LOB.');

1033:         $count = 0;
1034: 
1035:         while (!$lob->eof()) {

1039:                 throw $this->stream_error('Cannot read Oracle LOB.');
1040:             }
1041: 
1042:             $count += $this->write_all($stream, $chunk);
1043:         }
1044: 
1045:         return $count;
1046:     } // copy_value_to_stream
