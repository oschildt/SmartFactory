14: /**
15:  * Implements request limits, history expiry and retry delays for interchangeable storage backends.
16:  *
17:  * Subclasses supply history reading, persistence and an atomic execution boundary. The boundary
18:  * must protect the complete read/check/write operation for the same caller key across processes.
19:  * Caller keys are hashed once and passed explicitly to each storage hook. Rejected requests
20:  * are not recorded. Backends maintain expiry of inactive callers independently.
21:  * The interface remains the dependency and factory contract for concrete implementations.
22:  *
23:  * @author Oleg Schildt
24:  */
25: abstract class AbstractRequestThrottlingManager implements IRequestThrottlingManager
26: {
27:     /**
28:      * Stores maximum-count/period pairs applied simultaneously to each caller.
29:      *
30:      * @var list<array{0: positive-int, 1: positive-int}>
31:      *
32:      * @author Oleg Schildt
33:      */
34:     protected array $limits = self::DEFAULT_LIMITS;
35: 
36:     /**
37:      * Records whether initialization has successfully applied a policy.
38:      *
39:      * @var bool
40:      *
41:      * @author Oleg Schildt
42:      */
43:     protected bool $initialized = false;
44: 
45:     /**
46:      * Validates and applies rolling-window limits without changing shared request history.
47:      *
48:      * Subclasses should validate storage settings before calling this implementation and
49:      * commit their validated settings afterward, preserving the old configuration on failure.
50:      *
51:      * @param array{limits?: list<array{0: positive-int, 1: positive-int}>} $parameters
52:      * Count/period pairs containing positive integers. Omission selects DEFAULT_LIMITS;
53:      * an empty list disables limits. Other parameters are reserved for storage implementations.
54:      *
55:      * @return void
56:      * Applies a validated policy and marks the manager initialized; rejected settings change neither.
57:      *
58:      * @throws \InvalidArgumentException
59:      * If limits are not a list of exactly two positive integers per entry.
60:      *
61:      * @author Oleg Schildt
62:      */
63:     public function init(array $parameters): void
64:     {
65:         $limits = array_key_exists("limits", $parameters) ? $parameters["limits"] : self::DEFAULT_LIMITS;
66: 
67:         if (!is_array($limits) || !array_is_list($limits)) {
68:             throw new \InvalidArgumentException("The limits must be a list of maximum-count/period pairs.");
69:         }
70: 
71:         foreach ($limits as $limit) {
72:             if (
73:                 !is_array($limit)
74:                 || !array_is_list($limit)
75:                 || count($limit) !== 2
76:                 || !is_int($limit[0])
77:                 || !is_int($limit[1])
78:                 || $limit[0] <= 0
79:                 || $limit[1] <= 0
80:             ) {
81:                 throw new \InvalidArgumentException("Each limit must contain a positive integer count and period in seconds.");
82:             }
83:         }
84: 
85:         $this->limits = $limits;
86:         $this->initialized = true;
87:     } // init
88: 
89:     /**
90:      * Retrieves the policy applied to every caller key.
91:      *
92:      * @return list<array{0: positive-int, 1: positive-int}>
93:      * Returns count/period pairs, or an empty list when limits are disabled.
94:      *
95:      * @author Oleg Schildt
96:      */
97:     public function getLimits(): array
98:     {
99:         return $this->limits;
100:     } // getLimits

102:     /**
103:      * Reads the clock inside the storage's atomic execution boundary.
104:      *
105:      * @return float
106:      * Returns Unix time in seconds with subsecond precision.
107:      *
108:      * @author Oleg Schildt
109:      */
110:     protected function getCurrentTime(): float
111:     {
112:         return microtime(true);
113:     } // getCurrentTime

115:     /**
116:      * Removes timestamps outside the longest configured window for one caller.
117:      *
118:      * @param list<float> $timestamps
119:      * Validated, chronologically ordered Unix timestamps belonging to one caller.
120:      *
121:      * @param float $now
122:      * A finite, nonnegative reference time in Unix seconds.
123:      *
124:      * @return list<float>
125:      * Returns retained timestamps in their original order, preserving future entries after a clock rollback.
126:      * With disabled limits, returns the supplied timestamps unchanged.
127:      *
128:      * @author Oleg Schildt
129:      */
130:     protected function pruneRecords(array $timestamps, float $now): array
131:     {
132:         if ($this->limits === []) {
133:             return $timestamps;
134:         }
135: 
136:         $longest_period = max(array_column($this->limits, 1));
137:         $active = [];
138: 
139:         foreach ($timestamps as $timestamp) {
140:             if (round($now - $timestamp, 6) < $longest_period) {
141:                 $active[] = $timestamp;
142:             }
143:         }
144: 
145:         return $active;
146:     } // pruneRecords

148:     /**
149:      * Atomically reserves one request if every rolling-window limit permits it.
150:      *
151:      * @param string $key
152:      * A nonempty caller identity supplied by the request handler.
153:      *
154:      * @return int
155:      * Returns zero when admitted, or the longest required delay rounded up to whole seconds,
156:      * capped at PHP_INT_MAX. Rejected requests do not change stored history. Accepted requests
157:      * remove this caller's history older than the longest configured period. Other callers'
158:      * history and its background expiry remain the responsibility of the storage backend.
159:      * Future timestamps are retained if the system clock moves backwards.
160:      * Elapsed times use microsecond precision so fractional window boundaries expire exactly.
161:      *
162:      * @throws \InvalidArgumentException
163:      * If the caller key is empty.
164:      *
165:      * @throws \LogicException
166:      * If init() has not successfully initialized the manager.
167:      *
168:      * @throws \RuntimeException
169:      * If locking, the clock, reading, history validation or persistence fails.
170:      * Storage failures deny admission; the backend must release any acquired lock.
171:      *
172:      * @throws \Throwable
173:      * If a storage hook or overridden clock propagates another exception or error through atomic execution.
174:      *
175:      * @author Oleg Schildt
176:      */
177:     public function registerRequest(string $key): int
178:     {
179:         if ($key === "") {
180:             throw new \InvalidArgumentException("The throttling caller key must not be empty.");
181:         }
182: 
183:         if (!$this->initialized) {
184:             throw new \LogicException("The throttling manager must be initialized before use.");
185:         }
186: 
187:         if ($this->limits === []) {
188:             return 0;
189:         }
190: 
191:         $client = hash("sha256", $key);
192: 
193:         return $this->executeAtomically($client, function () use ($client): int {
194:             $now = round($this->getCurrentTime(), 6);
195: 
196:             if (!is_finite($now) || $now < 0) {
197:                 throw new \RuntimeException("The throttling clock returned an invalid timestamp.");
198:             }
199: 
200:             $timestamps = $this->pruneRecords($this->loadRecords($client), $now);
201:             $total = count($timestamps);
202:             $retry_after = 0;
203: 
204:             foreach ($this->limits as [$maximum, $period]) {
205:                 $count = 0;
206: 
207:                 foreach ($timestamps as $timestamp) {
208:                     if (round($now - $timestamp, 6) < $period) {
209:                         $count++;
210:                     }
211:                 }
212: 
213:                 if ($count >= $maximum) {
214:                     // Enough of the oldest entries must expire to leave a slot in this window.
215:                     $delay = ceil($period - round($now - $timestamps[$total - $maximum], 6));
216:                     $delay = $delay >= PHP_INT_MAX ? PHP_INT_MAX : max(1, (int) $delay);
217:                     $retry_after = max($retry_after, $delay);
218:                 }
219:             }
220: 
221:             if ($retry_after > 0) {
222:                 return $retry_after;
223:             }
224: 
225:             $timestamps[] = $now;
226: 
227:             // A backwards clock adjustment must not break chronological expiry calculations.
228:             sort($timestamps, SORT_NUMERIC);
229: 
230:             $this->saveRecords($client, $timestamps);
231: 
232:             return 0;
233:         });
234:     } // registerRequest

236:     /**
237:      * Executes one admission operation while protecting the caller's history against concurrent updates.
238:      *
239:      * Subclasses must invoke the callback exactly once after acquiring a lock or equivalent
240:      * transaction for this key, commit successful updates and release resources in a finally block.
241:      * Protection must include the first admission for a key with no stored history yet.
242:      * Shared-file implementations may lock the entire file to preserve unrelated callers' records.
243:      * Locking only loadRecords() and saveRecords() independently is insufficient: another caller
244:      * could reserve the same remaining slot between those operations.
245:      *
246:      * @param string $key
247:      * The lowercase SHA-256 caller hash used by every storage hook in this operation.
248:      *
249:      * @param callable(): int $operation
250:      * The complete clock/read/check/write operation to execute inside the protected boundary.
251:      *
252:      * @return int
253:      * Returns the callback's admission result unchanged.
254:      *
255:      * @throws \RuntimeException
256:      * If acquiring, committing or releasing the storage protection fails.
257:      *
258:      * @throws \Throwable
259:      * If the admission callback fails; its failure must propagate after resources are released.
260:      *
261:      * @author Oleg Schildt
262:      */
263:     abstract protected function executeAtomically(string $key, callable $operation): int;
264: 
265:     /**
266:      * Reads and validates one caller's retained history inside its atomic execution boundary.
267:      *
268:      * @param string $key
269:      * The lowercase SHA-256 caller hash supplied to executeAtomically().
270:      *
271:      * @return list<float>
272:      * Returns only this caller's timestamps, ordered from oldest to newest.
273:      * Equal timestamps represent separate requests and must remain separate entries.
274:      * Timestamps are finite, nonnegative Unix seconds with microsecond precision. Missing history
275:      * returns an empty array; malformed or unavailable existing history must not be treated as empty.
276:      *
277:      * @throws \RuntimeException
278:      * If history cannot be read or does not satisfy the storage schema and timestamp ordering.
279:      *
280:      * @author Oleg Schildt
281:      */
282:     abstract protected function loadRecords(string $key): array;
283: 
284:     /**
285:      * Persists one caller's updated history inside its atomic execution boundary.
286:      *
287:      * @param string $key
288:      * The lowercase SHA-256 caller hash supplied to executeAtomically().
289:      *
290:      * @param list<float> $timestamps
291:      * This caller's ordered timestamps, with expired history already removed.
292:      * Repeated timestamp values represent separate requests and must not be deduplicated.
293:      *
294:      * @return void
295:      * Replaces this caller's history and makes the update visible before atomic execution finishes.
296:      * Other callers' active history must be preserved. Inactive history may be expired by the backend.
297:      *
298:      * @throws \RuntimeException
299:      * If persistence fails; incomplete writes must not replace the last valid history.
300:      *
301:      * @author Oleg Schildt
302:      */
303:     abstract protected function saveRecords(string $key, array $timestamps): void;
