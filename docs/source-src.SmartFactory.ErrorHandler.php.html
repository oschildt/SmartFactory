313:     /**
314:      * Extracts a bounded stack while retaining class names, call types and application closures.
315:      * Only this handler's frames, trigger_error() and the library's registration
316:      * callback are skipped; unrelated methods named handleError remain visible.
317:      *
318:      * @param list<array{file?: string, line?: int, function?: string, class?: string, type?: string, args?: array<array-key, mixed>}> $btrace
319:      * Standard backtrace frames; values are omitted unless trace_arguments is enabled.
320:      *
321:      * @return string
322:      * Returns the formatted stack, limited to 64 frames and 65536 bytes.
323:      *
324:      * @author Oleg Schildt
325:      */
326:     protected function extract_call_stack(array $btrace): string
327:     {
328:         $lines = [];
329:         $size = 0;
330:         $registration_file = str_replace("\\", "/", dirname(__DIR__) . "/error_handler_init_inc.php");
331: 
332:         foreach (array_slice($btrace, 0, self::MAX_FRAMES) as $entry) {
333:             $file = str_replace("\\", "/", $entry["file"] ?? "");
334:             $function = $entry["function"] ?? "";
335:             $class = $entry["class"] ?? "";
336: 
337:             if (
338:                 $file === "" || $file === $registration_file ||
339:                 ($class === "" && $function === "trigger_error") ||
340:                 ($class !== "" && is_a($class, self::class, true))
341:             ) {
342:                 continue;
343:             }
344: 
345:             if (preg_match("/{closure:([^}{]+)}/", $function, $matches)) {
346:                 $function = "{closure:" . $this->trim_path($matches[1]) . "}";
347:             }
348: 
349:             $arguments = "";
350: 
351:             if (!empty($entry["args"])) {
352:                 $arguments = $this->trace_arguments
353:                     ? $this->make_arg_list($this->name_arguments($entry))
354:                     : "[arguments omitted]";
355:             }
356: 
357:             $call = $class . ($entry["type"] ?? "") . $function;
358:             $line = str_repeat("  ", count($lines)) . ($call === "" ? "" : $call . "(" . $arguments . ")") .
359:                 " [" . $this->trim_path($file) . ", " . ($entry["line"] ?? "line number undefined") . "]";
360:             $lines[] = $this->limit_message($line);
361:             $size += strlen($line);

363:             if ($size >= self::MAX_MESSAGE_BYTES) {
364:                 break;
365:             }
366:         }
367: 
368:         if (count($btrace) > self::MAX_FRAMES || $size >= self::MAX_MESSAGE_BYTES) {
369:             $lines[] = "... [call stack truncated]";
370:         }
371: 
372:         return $this->limit_message(implode(PHP_EOL, $lines));
373:     } // extract_call_stack

375:     /**
376:      * Truncates an argument preview after normalizing control characters to spaces.
377:      * Work is limited to a short input prefix even for very large strings.
378:      *
379:      * @param string|null $str
380:      * UTF-8 or binary input; invalid UTF-8 bytes are replaced, and null becomes "null".
381:      *
382:      * @param int $limit
383:      * Maximum preview length in Unicode characters; nonpositive limits omit all characters.
384:      *
385:      * @return string
386:      * Returns the preview followed by "..." if any input was omitted.
387:      *
388:      * @author Oleg Schildt
389:      */
390:     protected function truncate_argument(?string $str, int $limit): string
391:     {
392:         if ($str === null) {
393:             return "null";

396:         $limit = max(0, $limit);

399:         $truncated = strlen($str) > $limit * 4 + 4 || mb_strlen($prefix, "UTF-8") > $limit;

449:     /**
450:      * Renders one diagnostic value without invoking user-defined object conversion methods.
451:      *
452:      * @param mixed $value
453:      * Any PHP value, including closed resources and arrays containing recursive references.
454:      *
455:      * @param int $depth
456:      * Current array nesting level, starting at zero.
457:      *
458:      * @return string
459:      * Returns a bounded preview with strings quoted, explicit booleans and null,
460:      * resource markers, class names and masked sensitive array values.
461:      *
462:      * @author Oleg Schildt
463:      */
464:     protected function format_argument(mixed $value, int $depth = 0): string
465:     {
466:         if ($value instanceof \SensitiveParameterValue) {
467:             return "[REDACTED]";
468:         }
469: 
470:         if ($value === null) {
471:             return "null";
472:         }
473: 
474:         if (is_bool($value)) {
475:             return $value ? "true" : "false";
476:         }
477: 
478:         if (is_string($value)) {
479:             return json_encode($this->truncate_argument($value, 15), JSON_UNESCAPED_UNICODE | JSON_INVALID_UTF8_SUBSTITUTE);
480:         }
481: 
482:         if (is_int($value) || is_float($value)) {
483:             return var_export($value, true);
484:         }
485: 
486:         if (is_resource($value)) {
487:             return get_resource_type($value) . "#" . get_resource_id($value);
488:         }
489: 
490:         if (gettype($value) === "resource (closed)") {
491:             return "resource(closed)";
492:         }
493: 
494:         if (is_object($value)) {
495:             return get_class($value);
496:         }
497: 
498:         if ($depth >= self::MAX_ARRAY_DEPTH) {
499:             return "[...]";
500:         }
501: 
502:         $items = [];
503: 
504:         foreach ($value as $name => $item) {
505:             if (count($items) >= self::MAX_ARRAY_ITEMS) {
506:                 $items[] = "...";
507: 
508:                 break;
509:             }
510: 
511:             $label = $this->truncate_argument((string) $name, 40);
512:             $items[] = $label . "=" . ($this->is_sensitive_name($name) ? "[REDACTED]" : $this->format_argument($item, $depth + 1));
513:         }
514: 
515:         return "[" . implode(", ", $items) . "]";
516:     } // format_argument

518:     /**
519:      * Recognizes common credential names in parameter names and associative array keys.
520:      *
521:      * @param string|int $name
522:      * Parameter name, array key or positional index; numeric indices are not sensitive names.
523:      *

526:      *

529:     protected function is_sensitive_name(string|int $name): bool

537:     /**
538:      * Assigns available parameter names to positional arguments for credential masking.
539:      *
540:      * @param array{function?: string, class?: string, args?: array<array-key, mixed>} $entry
541:      * Standard backtrace frame; closures and unavailable internal functions may lack metadata.
542:      *
543:      * @return array<array-key, mixed>
544:      * Returns at most seventeen arguments for the bounded formatter. Sensitive variadic
545:      * parameters are masked for every supplied value, without reflecting argument objects.
546:      *
547:      * @author Oleg Schildt
548:      */
549:     protected function name_arguments(array $entry): array
550:     {
551:         $args = array_slice($entry["args"] ?? [], 0, self::MAX_ARGUMENTS + 1, true);
552:         $parameters = [];
553: 
554:         if (empty($entry["function"])) {
555:             return $args;
556:         }
557: 
558:         try {
559:             $reflection = isset($entry["class"])
560:                 ? new \ReflectionMethod($entry["class"], $entry["function"])
561:                 : new \ReflectionFunction($entry["function"]);
562:             $parameters = $reflection->getParameters();
563:         } catch (\ReflectionException) {
564:             // Generated closure names cannot be resolved through ReflectionFunction.
565:         }
566: 
567:         $named = [];
568:         $variadic = null;
569: 
570:         foreach ($args as $index => $arg) {
571:             $parameter = is_int($index) ? ($parameters[$index] ?? $variadic) : null;
572: 
573:             if ($parameter !== null && $parameter->isVariadic()) {
574:                 $variadic = $parameter;
575:             }
576: 
577:             $name = $parameter === null ? $index : $parameter->getName();
578:             $sensitive = $this->is_sensitive_name($name) ||
579:                 ($parameter !== null && $parameter->getAttributes(\SensitiveParameter::class) !== []);
580: 
581:             if ($sensitive) {
582:                 $arg = new \SensitiveParameterValue($arg);
583:             }
584: 
585:             // Keep every variadic value instead of overwriting a repeated parameter name.
586:             $key = array_key_exists($name, $named) ? $index : $name;
587:             $named[$key] = $arg;
588:         }
589: 
590:         return $named;
591:     } // name_arguments

593:     /**

595:      * URI query values and fragments are never included in diagnostics.
596:      *
597:      * @param string $message
598:      * Error or exception description supplied by the application or PHP.
599:      *
600:      * @param string $file
601:      * Source file where the error or exception originated.
602:      *
603:      * @param int $line
604:      * Source line reported by PHP or the exception.
605:      *
606:      * @param list<array{file?: string, line?: int, function?: string, class?: string, type?: string, args?: array<array-key, mixed>}> $trace
607:      * Standard backtrace frames without a synthetic header.
608:      *
609:      * @return string
610:      * Returns a diagnostic limited to 65536 bytes, including a sanitized request URI.
611:      *

615:     {

648:     /**
649:      * Appends a bounded diagnostic under an exclusive lock, falling back to PHP's server log.
650:      * Missing initialization and failed or incomplete file writes never imply stdout.
651:      *
652:      * @param string $etype
653:      * Human-readable PHP error category or an unknown numeric error code as a string.
654:      *
655:      * @param string $message
656:      * Formatted diagnostic; messages larger than 65536 bytes are truncated.
657:      *
658:      * @return bool
659:      * Returns true when tracing is disabled, stdout was explicitly used, the entire
660:      * file entry was written, or error_log() accepted the fallback. Returns false
661:      * if both the primary destination and fallback failed; records attempted write results.
662:      *
663:      * @author Oleg Schildt
664:      */
665:     protected function trace_message(string $etype, string $message): bool
666:     {
667:         if (!$this->traceActive()) {
668:             return true;
669:         }
670: 
671:         $message = date("Y-m-d H:i:s") . PHP_EOL .
672:             "------------------------------------" . PHP_EOL .
673:             "### " . $etype . " ###" . PHP_EOL . PHP_EOL .
674:             $this->limit_message($message) . PHP_EOL .
675:             "------------------------------------" . PHP_EOL . PHP_EOL . PHP_EOL;

678:         try {

681:                 $this->last_trace_succeeded = true;
682: 
683:                 return true;
684:             }
685: 
686:             if ($this->log_path !== null && $this->log_path !== "stdout") {
687:                 clearstatcache(true, $file);
688: 
689:                 if (!is_link($file) && (!file_exists($file) || is_file($file))) {
690:                     $written = @file_put_contents($file, $message, FILE_APPEND | LOCK_EX);

692:                     if ($written === strlen($message)) {
693:                         $this->last_trace_succeeded = true;

695:                         return true;
696:                     }
697:                 }
698:             }
699:         } catch (\Throwable) {
700:             // A logging failure must not replace the original application error.
701:         }

703:         $reason = $this->log_path === null
704:             ? "The trace file is not specified."
705:             : ($this->log_path === "stdout" ? "Web stdout diagnostics are disabled." : "Could not write the complete trace entry to '" . $file . "'.");
706: 
707:         try {
708:             $this->last_trace_succeeded = @error_log($reason . PHP_EOL . $message);
709:         } catch (\Throwable) {

717:      * Handles reportable PHP errors, stores their text and notifies programming-warning listeners.
718:      * Masked errors, including ordinary warnings suppressed with @, are ignored.
719:      * Backtraces are collected only while tracing is enabled; recursive callbacks are ignored.
720:      *
721:      * @param int $errno
722:      * PHP error level, checked against the active error_reporting() mask.
723:      *
724:      * @param string $errstr
725:      * Original error message to store and log.

727:      * @param string $errfile
728:      * Source file where the error occurred.

730:      * @param int $errline
731:      * Source line where the error occurred.
732:      *

735:      *
736:      * @throws \Throwable
737:      * If resolving the event manager or executing a php_error listener throws;
738:      * the recursion guard is restored before the failure is propagated.
739:      *
740:      * @author Oleg Schildt
741:      */
742:     public function handleError(int $errno, string $errstr, string $errfile, int $errline): void
743:     {

745:             return;
746:         }
747:

771:     /**
772:      * Logs a caught throwable with its class, code, source and bounded previous-exception chain.
773:      * Explicit exception logging is independent of the PHP error_reporting() mask.
774:      *
775:      * @param \Throwable $ex
776:      * Exception or error to record; previous exceptions are followed for at most eight entries.
777:      *
778:      * @param int $errno
779:      * PHP category used to label the log entry, including E_USER_DEPRECATED or unknown codes.
780:      *
781:      * @return void
782:      * Stores the outer message and logs the bounded chain only when tracing is enabled.
783:      *
784:      * @author Oleg Schildt
785:      */
786:     public function handleException(\Throwable $ex, int $errno): void
787:     {
788:         $this->setLastError($ex->getMessage());
789: 
790:         if (!$this->traceActive()) {
791:             return;
792:         }
793: 
794:         $parts = [];
795:         $size = 0;
796:         $current = $ex;
797: 
798:         for ($index = 0; $current !== null && $index < self::MAX_EXCEPTIONS; $index++) {
799:             $description = ($index === 0 ? "" : "Caused by: ") . get_class($current) .
800:                 " (code " . $current->getCode() . "): " . $this->limit_message($current->getMessage());
801:             $part = $this->format_diagnostic($description, $current->getFile(), $current->getLine(), $current->getTrace());
802:             $parts[] = $part;
803:             $size += strlen($part);
804:             $current = $current->getPrevious();
805: 
806:             if ($size >= self::MAX_MESSAGE_BYTES) {
807:                 break;
808:             }
809:         }
810: 
811:         if ($current !== null) {
812:             $parts[] = "... [exception chain truncated]";
813:         }
814: 
815:         $this->trace_message(self::ERROR_TYPES[$errno] ?? (string) $errno, $this->limit_message(implode(PHP_EOL, $parts)));
816:     } // handleException
