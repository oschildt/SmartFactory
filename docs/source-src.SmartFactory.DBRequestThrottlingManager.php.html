16: /**
17:  * Persists one caller's request history in one database row under a transactional logical-key lock.
18:  *
19:  * Uses DBWorker::lock_record(), including keys without a stored row, and save_records() for bound
20:  * upserts. The configured database needs the target table and DBWorker's dbworker_locks service table.
21:  * Each admission owns a READ COMMITTED transaction so application rollbacks cannot refund requests.
22:  * Connected clones preserve the caller's prepared statement while sharing its native connection.
23:  *
24:  * @link markdown:DBRequestThrottlingManager.md Important notes
25:  *
26:  * @author Oleg Schildt

75:     /**
76:      * Configures the database worker, storage table and inherited rolling-window limits.
77:      *
78:      * @param array{dbworker: DBWorker, table: string, limits?: list<array{0: positive-int, 1: positive-int}>} $parameters
79:      * The worker and an unquoted table name, optionally qualified by schema or database.
80:      * The table must contain client_key, request_times and expires_at columns as documented
81:      * in DBRequestThrottlingManager.md. Omitted limits use DEFAULT_LIMITS; an empty list disables limits.
82:      *
83:      * @return void
84:      * Applies validated configuration without connecting, creating tables or clearing stored history.
85:      * Invalid initialization preserves the previous configuration.

142:     /**
143:      * Serializes one caller's complete admission operation inside an owned READ COMMITTED transaction.
144:      *
145:      * @param string $key
146:      * The lowercase SHA-256 caller hash supplied by the inherited admission algorithm.
147:      *
148:      * @param callable(): int $operation
149:      * The complete clock/read/check/write operation to run after acquiring the caller's logical-key lock.
150:      *
151:      * @return int
152:      * Returns the admission result only after committing and releasing the logical-key lock.
153:      *
154:      * @throws \LogicException
155:      * If no worker is configured, admission is nested, or the configured connection already has
156:      * a managed transaction. Use a separate connection when application work is already transactional.
157:      *
158:      * @throws \InvalidArgumentException
159:      * If the worker's schema produces an unsupported qualified table name.
160:      *
161:      * @throws \Throwable
162:      * If connecting, cloning, starting, locking, the callback, committing or statement cleanup fails.
163:      * DBWorkerException failures are wrapped in RuntimeException with the original cause so handlers report 500.
164:      * Owned transactions are rolled back after failures when still active. A rollback or cleanup failure
165:      * does not replace the original failure; a connection whose rollback failed needs caller recovery.
166:      *
167:      * @author Oleg Schildt
168:      */
169:     protected function executeAtomically(string $key, callable $operation): int
170:     {
171:         try {
172:             return $this->withTransaction($key, $operation);
173:         } catch (DBWorkerException $exception) {

175:             throw new \RuntimeException("The database throttling storage operation failed.", 0, $exception);
176:         }

179:     /**
180:      * Owns a READ COMMITTED transaction and its selected caller's logical-key lock.
181:      *
182:      * @param string $key
183:      * The caller hash identifying the admission operation.
184:      *
185:      * @param callable(): int $operation
186:      * The complete admission callback, invoked only after locking this key.
187:      *
188:      * @return int
189:      * Returns the result after committing the owned transaction.
190:      *
191:      * @throws \LogicException
192:      * If configuration is missing, admission is nested or an application transaction is already active.
193:      *
194:      * @throws \Throwable
195:      * If connection setup, qualification, transaction operations, locking, the callback or cleanup fails.
196:      * A failed owned transaction is rolled back when still active without ending an application transaction.
197:      *
198:      * @author Oleg Schildt
199:      */
200:     private function withTransaction(string $key, callable $operation): int
201:     {
202:         if ($this->dbworker === null || $this->table === "") {
203:             throw new \LogicException("The database throttling manager is not initialized.");
204:         }
205: 
206:         if ($this->active_worker !== null) {
207:             throw new \LogicException("Nested throttling admission operations are not supported.");
208:         }
209: 
210:         if (!$this->dbworker->is_connected()) {
211:             $this->dbworker->connect();
212:         }
213: 
214:         $worker = $this->dbworker->create_clone();
215: 
216:         if ($worker->get_transaction_isolation_level() !== null) {
217:             throw new \LogicException("Database throttling requires a connection without an active managed transaction.");
218:         }
219: 
220:         $table = $worker->qualify_name_with_schema($this->table);
221: 
222:         $this->validateTable($table);
223: 
224:         $worker->start_transaction(DBWorker::ISOLATION_READ_COMMITTED);
225: 
226:         $this->active_worker = $worker;
227:         $this->active_key = $key;
228:         $this->active_table = $table;
229:         $failure = null;
230: 
231:         try {
232:             // The logical lock also protects simultaneous first admissions before the target row exists.
233:             $worker->lock_record($table, ["client_key" => $key]);
234: 
235:             $result = $operation();
236: 
237:             $worker->commit_transaction();
238: 
239:             return $result;
240:         } catch (\Throwable $exception) {
241:             $failure = $exception;
242: 
243:             if ($worker->get_transaction_isolation_level() !== null) {
244:                 try {
245:                     $worker->rollback_transaction();
246:                 } catch (\Throwable) {
247:                     // Preserve the admission failure; the caller must recover an unusable connection.
248:                 }
249:             }
250: 
251:             throw $exception;
252:         } finally {
253:             $this->active_worker = null;
254:             $this->active_key = null;
255:             $this->active_table = "";
256: 
257:             try {
258:                 $worker->free_prepared_query();
259:             } catch (\Throwable $cleanup_exception) {
260:                 if ($failure === null) {
261:                     throw $cleanup_exception;
262:                 }
263:             }
264:         }
265:     } // withTransaction

290:     /**
291:      * Loads and validates the JSON timestamp list stored for one caller.
292:      *
293:      * @param string $key
294:      * The lowercase SHA-256 caller hash supplied to executeAtomically().
295:      *
296:      * @return list<float>
297:      * Returns ordered finite, nonnegative timestamps, preserving duplicates, or an empty list
298:      * when the caller has no row. Existing null, empty, malformed or non-array payloads fail closed.
299:      *
300:      * @throws \LogicException
301:      * If the key does not belong to the active admission transaction.
302:      *
303:      * @throws \RuntimeException
304:      * If the caller key is not unique or stored history is invalid.
305:      *
306:      * @throws \Throwable
307:      * If preparing, binding, reading, converting a CLOB or releasing the statement fails.
308:      *
309:      * @author Oleg Schildt
310:      */
311:     protected function loadRecords(string $key): array
312:     {
313:         $worker = $this->activeWorker($key);
314:         $failure = null;
315: 
316:         try {
317:             $worker->prepare_query("SELECT request_times FROM " . $this->active_table . " WHERE client_key = ?");
318: 
319:             $worker->execute_prepared_query($key);
320: 
321:             if (!$worker->fetch_row()) {
322:                 return [];
323:             }
324: 
325:             $json = $worker->field_by_name("request_times", DBWorker::DB_STRING);
326: 
327:             if ($worker->fetch_row()) {
328:                 throw new \RuntimeException("The throttling caller key is not unique.");
329:             }
330:         } catch (\Throwable $exception) {
331:             $failure = $exception;
332: 
333:             throw $exception;
334:         } finally {
335:             try {
336:                 $worker->free_prepared_query();
337:             } catch (\Throwable $cleanup_exception) {
338:                 if ($failure === null) {
339:                     throw $cleanup_exception;
340:                 }
341:             }
342:         }
343: 
344:         if (!is_string($json) || $json === "") {
345:             throw new \RuntimeException("The stored throttling history is empty or is not text.");
346:         }
347: 
348:         try {
349:             $values = json_decode($json, false, 512, JSON_THROW_ON_ERROR);
350:         } catch (\JsonException $exception) {
351:             throw new \RuntimeException("The stored throttling history is invalid JSON.", 0, $exception);
352:         }
353: 
354:         if (!is_array($values)) {
355:             throw new \RuntimeException("The stored throttling history must be a JSON array.");
356:         }
357: 
358:         $timestamps = [];
359:         $previous = -1.0;
360: 
361:         foreach ($values as $value) {
362:             if ((!is_int($value) && !is_float($value)) || !is_finite((float) $value) || $value < 0 || $value < $previous) {
363:                 throw new \RuntimeException("The stored throttling timestamps are invalid or out of order.");
364:             }
365: 
366:             $timestamp = (float) $value;
367:             $timestamps[] = $timestamp;
368:             $previous = $timestamp;
369:         }
370: 
371:         return $timestamps;
372:     } // loadRecords

374:     /**
375:      * Upserts only the selected caller's history and its indexed expiry timestamp.
376:      *
377:      * @param string $key
378:      * The lowercase SHA-256 caller hash supplied to executeAtomically().
379:      *
380:      * @param list<float> $timestamps
381:      * This caller's nonempty ordered timestamp list prepared by the common admission algorithm.
382:      * Equal timestamps represent separate requests and are preserved.
383:      *
384:      * @return void
385:      * Saves the caller's JSON/CLOB payload and expiry, preserving every other caller.
386:      * Expiry is the newest retained timestamp plus the longest configured window.
387:      *
388:      * @throws \LogicException
389:      * If the key does not belong to the active admission transaction or the list is unexpectedly empty.
390:      *
391:      * @throws \Throwable
392:      * If JSON encoding, CLOB creation or the bound transactional upsert fails.
393:      *
394:      * @author Oleg Schildt
395:      */
396:     protected function saveRecords(string $key, array $timestamps): void
397:     {
398:         $worker = $this->activeWorker($key);
399: 
400:         if ($timestamps === []) {
401:             throw new \LogicException("An admitted throttling request must contain at least one timestamp.");
402:         }
403: 
404:         $json = json_encode($timestamps, JSON_THROW_ON_ERROR | JSON_PRESERVE_ZERO_FRACTION);
405:         $expires_at = max($timestamps) + max(array_column($this->limits, 1));
406:         $record = [
407:             "client_key" => $key,
408:             "request_times" => new ClobValue($json),
409:             "expires_at" => sprintf("%.6F", $expires_at),
410:         ];
411: 
412:         $worker->save_records($record, [
413:             "table" => $this->active_table,
414:             "fields" => [
415:                 "client_key" => DBWorker::DB_STRING,
416:                 "request_times" => DBWorker::DB_AS_IS,
417:                 "expires_at" => DBWorker::DB_NUMBER,
418:             ],
419:             "array_dimension_keys" => [],
420:             "record_key_fields" => ["client_key"],
421:         ]);
422:     } // saveRecords
