15: /**
16:  * Stores application-wide settings centrally in a database as a JSON document.
17:  *
18:  * Central storage allows an application distributed across several servers to share
19:  * the same settings. A stable record key identifies the document. Saves lock that key,
20:  * read the latest document and merge only explicitly edited parameters in one transaction.
21:  * Concurrent edits to other parameters survive; the last save wins for the same parameter.
22:  * Values are cached in this object; call loadSettings() to refresh a long-lived instance.
23:  * Contextual validation supports separate forms, but drafts are not stored between requests.
24:  *
25:  * @see ConfigSettingsManager
26:  * @see UserSettingsManager
27:  *
28:  * @author Oleg Schildt
29:  */

68:     /**
69:      * Configures database-backed settings without loading them immediately.
70:      *
71:      * @param array{dbworker: DBWorker, settings_table: string, settings_column: string, settings_id_column?: string, settings_id?: int, no_own_transaction?: bool} $parameters
72:      * The worker, table and JSON column. The unique, non-generated integer key defaults
73:      * to id = 1. A missing document is created on the first save. When no_own_transaction
74:      * is true the caller must already have started a READ COMMITTED transaction.
75:      * The settings table must provide the configured unique document key.
76:      *
77:      * @return void
78:      * Discards the previous snapshot and unsaved edits.
79:      *
80:      * @throws \InvalidArgumentException
81:      * If the worker, identifier names or document key have invalid types or values,
82:      * or the key column and JSON column have the same name.
83:      *
84:      * @author Oleg Schildt
85:      */
86:     public function init(array $parameters): void
87:     {
88:         $worker = $parameters['dbworker'] ?? null;
89:         $table = $parameters['settings_table'] ?? null;
90:         $column = $parameters['settings_column'] ?? null;
91:         $key = $parameters['settings_id_column'] ?? 'id';
92:         $id = $parameters['settings_id'] ?? 1;
93: 
94:         if (!$worker instanceof DBWorker || !is_string($table) || !is_string($column)
95:             || !is_string($key) || !is_int($id) || strcasecmp($column, $key) === 0) {
96:             throw new \InvalidArgumentException("Invalid application settings configuration.");
97:         }
98: 
99:         $this->validateIdentifier($table, true);
100: 
101:         $this->validateIdentifier($column);
102: 
103:         $this->validateIdentifier($key);
104: 
105:         $this->settings_table = $table;
106:         $this->settings_column = $column;
107:         $this->settings_id_column = $key;
108:         $this->settings_id = $id;
109: 
110:         $this->setDBWorker($worker, (bool) ($parameters['no_own_transaction'] ?? false));

131:     /**
132:      * Reads one keyed JSON document through a bound query.
133:      *
134:      * @param DBWorker $worker
135:      * The connected worker, optionally holding the document's logical key lock.
136:      *
137:      * @return array<string, mixed>
138:      * Returns the decoded document, or an empty array for a missing row or NULL payload.
139:      *
140:      * @throws \UnexpectedValueException
141:      * If the JSON value is not an object or array, or duplicate document keys are found.
142:      *
143:      * @throws \Throwable
144:      * If preparing, executing, fetching or decoding the document fails.
145:      *
146:      * @author Oleg Schildt
147:      */
148:     protected function readJSON(DBWorker $worker): array
149:     {
150:         try {
151:             $worker->prepare_query('SELECT ' . $this->settings_column . ' FROM ' . $this->settings_table
152:                 . ' WHERE ' . $this->settings_id_column . ' = ?');
153: 
154:             $worker->execute_prepared_query($this->settings_id);
155: 
156:             if (!$worker->fetch_row()) {
157:                 return [];
158:             }
159: 
160:             $json = $worker->field_by_name($this->settings_column, DBWorker::DB_STRING);
161: 
162:             if ($worker->fetch_row()) {
163:                 throw new \UnexpectedValueException("The application settings key is not unique.");
164:             }
165:         } finally {
166:             $worker->free_prepared_query();
167:         }
168: 
169:         if ($json === null || $json === '') {
170:             return [];
171:         }
172: 
173:         $data = json_decode($json, true, 512, JSON_THROW_ON_ERROR);
174: 
175:         if (!is_array($data)) {
176:             throw new \UnexpectedValueException("Application settings JSON must contain an object or array.");
177:         }
178: 
179:         return $data;
180:     } // readJSON

205:     /**
206:      * Locks the document key, merges local edits and saves through save_records().
207:      *
208:      * @param array<string, mixed> &$data
209:      * The local snapshot, replaced with the merged document only after success.
210:      *
211:      * @return void
212:      * Creates a missing keyed document and preserves unrelated concurrent edits.
213:      *
214:      * @throws \Throwable
215:      * If initialization, transaction setup, locking, reading, encoding or saving fails.
216:      * External transactions remain the caller's responsibility and must be rolled back on failure.
217:      *
218:      * @see ApplicationSettingsManager::loadJSON()
219:      *
220:      * @author Oleg Schildt
221:      */
222:     protected function saveJSON(array &$data): void
223:     {
224:         $this->validateParameters();
225: 
226:         $data = $this->withTransaction(function (DBWorker $worker) use ($data): array {
227:             // Lock before reading, including when the document has not been created yet.
228:             $worker->lock_record($this->settings_table, [$this->settings_id_column => $this->settings_id]);
229: 
230:             $merged = $this->readJSON($worker);
231: 
232:             foreach ($this->changed as $name => $_) {
233:                 $merged[$name] = $data[$name];
234:             }
235: 
236:             $json = json_encode($merged, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE
237:                 | JSON_PRESERVE_ZERO_FRACTION | JSON_THROW_ON_ERROR);
238:             $record = [$this->settings_id_column => $this->settings_id,
239:                 $this->settings_column => new ClobValue($json)];
240: 
241:             $worker->save_records($record, [
242:                 'table' => $this->settings_table,
243:                 'fields' => [$this->settings_id_column => DBWorker::DB_NUMBER,
244:                     $this->settings_column => DBWorker::DB_AS_IS],
245:                 'array_dimension_keys' => [],
246:                 'record_key_fields' => [$this->settings_id_column]
247:             ]);
248: 
249:             return $merged;
250:         });
251:     } // saveJSON
