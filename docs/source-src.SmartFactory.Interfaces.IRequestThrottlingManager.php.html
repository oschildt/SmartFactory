12: /**
13:  * Atomically enforces every configured rolling-window limit for an opaque caller key.
14:  *
15:  * Storage implementations may use files, databases or Redis. Checking limits and recording
16:  * an allowed request must be one atomic operation across all processes sharing the storage.
17:  * A limit is [maximum request count, period in seconds]. All limits apply simultaneously
18:  * to the interval (current time - period, current time]. Rejected requests are not recorded.
19:  * Clock rollbacks may retain future records conservatively instead of resetting budgets.
20:  * The abstract base class provides the shared algorithm for backends that only
21:  * need to implement history reading, persistence and atomic execution.
22:  *
23:  * @see \SmartFactory\AbstractRequestThrottlingManager
24:  *
25:  * @author Oleg Schildt
26:  */
27: interface IRequestThrottlingManager extends IInitable
28: {
29:     /**
30:      * Provides the demo admission policy when no explicit limits are supplied.
31:      *
32:      * @var list<array{0: positive-int, 1: positive-int}>
33:      *
34:      * @author Oleg Schildt
35:      */
36:     public const array DEFAULT_LIMITS = [[3, 1], [15, 10], [1550, 60], [10000, 500]];
37: 
38:     /**
39:      * Initializes the throttling policy and implementation-specific storage settings.
40:      *
41:      * @param array{limits?: list<array{0: positive-int, 1: positive-int}>} $parameters
42:      * The limits, using positive integer counts and periods in seconds. Omission selects
43:      * DEFAULT_LIMITS; an empty list disables admission limits. Implementations may require
44:      * additional entries for their storage connection or file path.
45:      *
46:      * @return void
47:      * Applies the policy without erasing existing shared request history.
48:      *
49:      * @throws \InvalidArgumentException
50:      * If a limit is not an integer pair with positive values or storage settings are invalid.
51:      *
52:      * @throws \RuntimeException
53:      * If the implementation initializes its storage eagerly and that storage is unavailable.
54:      *
55:      * @author Oleg Schildt
56:      */
57:     public function init(array $parameters): void;
58: 
59:     /**
60:      * Retrieves all limits currently enforced by the manager.
61:      *
62:      * @return list<array{0: positive-int, 1: positive-int}>
63:      * Returns count/period pairs, or an empty list when limits are disabled.
64:      *
65:      * @author Oleg Schildt
66:      */
67:     public function getLimits(): array;
68: 
69:     /**
70:      * Atomically checks every window and records the request only when all windows allow it.
71:      *
72:      * @param string $key
73:      * A nonempty opaque identity, such as a normalized IP address, user ID, API-key identifier
74:      * or one shared key for a global budget. The same key shares limits across API actions.
75:      *
76:      * @return int
77:      * Returns zero when admitted, or a positive Retry-After delay rounded up to whole seconds
78:      * until every limit permits another request, assuming no other requests are admitted meanwhile.
79:      * A rejection must not change request history.
80:      *
81:      * @throws \InvalidArgumentException
82:      * If the caller key is empty.
83:      *
84:      * @throws \LogicException
85:      * If the implementation requires initialization and has not been initialized.
86:      *
87:      * @throws \RuntimeException
88:      * If shared history cannot be read, locked, validated or updated; admission must not be granted.
89:      *
90:      * @author Oleg Schildt
91:      */
92:     public function registerRequest(string $key): int;
93: } // IRequestThrottlingManager
