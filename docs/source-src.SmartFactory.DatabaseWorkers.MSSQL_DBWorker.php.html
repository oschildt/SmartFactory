13: /**
14:  * Database worker for Microsoft SQL Server using the sqlsrv extension.
15:  *
16:  * Uses UTF-8 connections with Multiple Active Result Sets disabled.
17:  * All queries use forward-only cursors without buffering the complete result set.
18:  * An unread forward-only result must be exhausted or freed before another request uses the connection.

20:  * from UTF-16LE without splitting surrogate pairs.
21:  *
22:  * SQL Server has no native read-only transaction mode, so the read_only option is rejected.
23:  *
24:  * @see DBWorker

205:     /**
206:      * Establishes the database connection using the initialized settings.
207:      *
208:      * Uses UTF-8 and string date values with Multiple Active Result Sets disabled.
209:      * The read_only option is rejected because SQL Server cannot enforce a read-only transaction.
210:      * An already connected worker keeps its existing connection.
211:      *
212:      * @return void
213:      * Does not return a value.
214:      *
215:      * @throws DBWorkerException
216:      * If connection settings are incomplete, read_only is enabled, the isolation level is invalid,
217:      * or connection or session initialization fails.
218:      *
219:      * @see DBWorker::init()
220:      * @see MSSQL_DBWorker::is_connected()
221:      * @see DBWorker::close_connection()
222:      *
223:      * @author Oleg Schildt
224:      */
225:     public function connect(): void
226:     {
227:         // Reuse an attached connection and validate the settings before creating a new native handle.
228:         if ($this->is_connected()) {
229:             return;
230:         }
231: 
232:         if ($this->db_server === '' || $this->db_user === '') {
233:             throw new DBWorkerException('Connection data is incomplete.', self::ERR_CONNECTION_DATA_INCOMPLETE);
234:         }
235: 
236:         if ($this->read_only) {
237:             throw new DBWorkerException(
238:                 'SQL Server has no read-only transaction mode; enforce read-only access using database permissions.',
239:                 self::ERR_WRONG_DATA_FORMAT
240:             );
241:         }
242: 
243:         $server = $this->db_server . ($this->db_port === '' ? '' : ',' . (int) $this->db_port);
244:         $config = [
245:             'UID' => $this->db_user,
246:             'PWD' => $this->db_password,
247:             'CharacterSet' => 'UTF-8',
248:             'ReturnDatesAsStrings' => true,
249:             'MultipleActiveResultSets' => false,
250:             'LoginTimeout' => (int) ($this->options['connect_timeout'] ?? 20)
251:         ];
252: 
253:         if ($this->db_name !== '') {
254:             $config['Database'] = $this->db_name;
255:         }
256: 
257:         if (array_key_exists('encrypt', $this->options)) {
258:             $config['Encrypt'] = (bool) $this->options['encrypt'];
259:         }
260: 
261:         if (array_key_exists('trust_server_certificate', $this->options)) {
262:             $config['TrustServerCertificate'] = (bool) $this->options['trust_server_certificate'];
263:         }
264: 
265:         $db = @sqlsrv_connect($server, $config);
266: 
267:         if ($db === false) {
268:             throw new DBWorkerException($this->errors(), self::ERR_CONNECTION_FAILED);
269:         }
270: 
271:         $this->connection_state = new DBConnectionState($db, static fn ($db) => sqlsrv_close($db));
272: 
273:         if (isset($this->options['isolation_level'])) {
274:             $level = strtoupper((string) $this->options['isolation_level']);
275: 
276:             if (!in_array(
277:                     $level,
278:                     ['READ UNCOMMITTED', 'READ COMMITTED', 'REPEATABLE READ', 'SNAPSHOT', 'SERIALIZABLE'],
279:                     true
280:                 )) {
281:                 $this->connection_state = null;
282: 
283:                 throw new DBWorkerException('Invalid transaction isolation level.', self::ERR_WRONG_DATA_FORMAT);
284:             }
285: 
286:             $result = @sqlsrv_query($db, 'SET TRANSACTION ISOLATION LEVEL ' . $level);
287: 
288:             if ($result === false) {
289:                 $message = $this->errors();
290:                 $this->connection_state = null;
291: 
292:                 throw $this->query_error($message);
293:             }
294: 
295:             sqlsrv_free_stmt($result);
296:         }
297:     } // connect

339:     /**
340:      * Appends an identity query to an eligible SQL Server INSERT batch.
341:      *
342:      * The internal result is identified by a reserved column alias and consumed by accept_statement().
343:      *
344:      * @param string $sql
345:      * The SQL statement to inspect.
346:      *
347:      * @return string
348:      * Returns SQL with SCOPE_IDENTITY() in the same batch, or the original SQL when OUTPUT is already present.
349:      *

351:      */
352:     private function with_identity(string $sql): string
353:     {
354:         // Capture an identifier from the INSERT itself instead of consulting unrelated session sequence state.
355:         $this->captures_identity = $this->insert_table($sql) !== null && !preg_match('/\bOUTPUT\b/i', $sql);
356: 
357:         return $this->captures_identity ? rtrim($sql, " \t\r\n;")
358:             . '; SELECT CAST(SCOPE_IDENTITY() AS bigint) AS smartfactory_insert_id' : $sql;

406:      *
407:      * @return void

417:     protected function native_prepare(string $sql): void
418:     {
419:         // Keep native statement preparation separate from public placeholder compilation.
420:         $this->prepared_sql = $this->with_identity($sql);
421:         $this->bound_values = array_fill(0, count($this->compiled_query->names), null);
422:         $this->native_parameters = [];
423: 
424:         foreach ($this->bound_values as $index => &$value) {
425:             $this->native_parameters[$index] = [&$value, SQLSRV_PARAM_IN];
426:         }
427: 
428:         unset($value);
429: 
430:         $options = $this->statement_options();
431:         $stmt = @sqlsrv_prepare(
432:             $this->connection_state->handle,
433:             $this->prepared_sql,
434:             $this->native_parameters,
435:             $options
436:         );
437: 
438:         if ($stmt === false) {
439:             throw $this->query_error($this->errors());
440:         }
441: 
442:         $this->statement = $stmt;
443:         $this->parameter_signature = null;
444:     } // native_prepare

446:     /**
447:      * Binds validated values and executes the native prepared statement.
448:      *
449:      * Uses explicit PHP/SQL parameter types and re-prepares the native statement when its
450:      * type signature changes. Caller-owned stream parameters are copied into temporary
451:      * seekable streams, sent before result processing, and released in the cleanup path.
452:      *
453:      * @param list<scalar|null|BlobValue|ClobValue> $values
454:      * The normalized values in SQL occurrence order, including any explicit LOB wrappers.
455:      *
456:      * @return void
457:      * Does not return a value.
458:      *
459:      * @throws DBWorkerException
460:      * If binding, LOB processing, execution or result completion fails, including when
461:      * another worker has a pending forward-only result on the shared connection.
462:      *
463:      * @see DBWorker::execute_prepared_query()
464:      *
465:      * @author Oleg Schildt
466:      */
467:     protected function native_execute(array $values): void
468:     {
469:         // Bind stable per-position values so repeated application names cannot share a loop variable by accident.
470:         $temporary = [];
471:         $converted = [];
472:         $types = [];
473: 
474:         try {
475:             foreach ($values as $index => $value) {
476:                 if ($value instanceof BlobValue || $value instanceof ClobValue) {
477:                     $binary = $value instanceof BlobValue;
478: 
479:                     if (is_resource($value->value)) {
480:                         $stream = tmpfile();
481: 
482:                         if ($stream === false) {
483:                             throw $this->stream_error('Cannot spool a SQL Server stream parameter.');
484:                         }
485: 
486:                         $temporary[] = $stream;
487: 
488:                         $this->write_chunks(
489:                             $value->value,
490:                             fn (string $chunk) => $this->write_all($stream, $chunk),
491:                             !$binary
492:                         );
493: 
494:                         rewind($stream);
495: 
496:                         $converted[$index] = $stream;
497:                         $php = SQLSRV_PHPTYPE_STREAM($binary ? SQLSRV_ENC_BINARY : 'UTF-8');
498:                     } else {
499:                         $converted[$index] = $value->value;
500:                         $php = SQLSRV_PHPTYPE_STRING($binary ? SQLSRV_ENC_BINARY : 'UTF-8');
501:                     }
502: 
503:                     $types[$index] = [$php, $binary ? SQLSRV_SQLTYPE_VARBINARY('max') : SQLSRV_SQLTYPE_NVARCHAR('max')];
504:                 } elseif (is_int($value) || is_bool($value)) {
505:                     $converted[$index] = (int) $value;
506:                     $types[$index] = [SQLSRV_PHPTYPE_INT, SQLSRV_SQLTYPE_BIGINT];
507:                 } elseif (is_float($value)) {
508:                     $converted[$index] = $value;
509:                     $types[$index] = [SQLSRV_PHPTYPE_FLOAT, SQLSRV_SQLTYPE_FLOAT];
510:                 } else {
511:                     $converted[$index] = $value;
512:                     $types[$index] = [
513:                         $value === null ? SQLSRV_PHPTYPE_NULL : SQLSRV_PHPTYPE_STRING('UTF-8'),
514:                         SQLSRV_SQLTYPE_NVARCHAR('max')
515:                     ];
516:                 }
517:             }
518: 
519:             // Re-prepare only when native parameter types change; otherwise update the existing referenced values.
520:             $signature = serialize($types);
521: 
522:             if ($signature !== $this->parameter_signature) {
523:                 if (is_resource($this->statement)) {
524:                     sqlsrv_free_stmt($this->statement);
525:                 }
526: 
527:                 $this->statement = null;
528:                 $this->parameter_signature = null;
529:                 $this->bound_values = $converted;
530:                 $this->native_parameters = [];
531: 
532:                 foreach ($types as $index => [$php, $sql]) {
533:                     $this->native_parameters[$index] = [&$this->bound_values[$index], SQLSRV_PARAM_IN, $php, $sql];
534:                 }
535: 
536:                 $options = $this->statement_options();
537:                 $options['SendStreamParamsAtExec'] = false;
538:                 $stmt = @sqlsrv_prepare(
539:                     $this->connection_state->handle,
540:                     $this->prepared_sql,
541:                     $this->native_parameters,
542:                     $options
543:                 );
544: 
545:                 if ($stmt === false) {
546:                     throw $this->query_error($this->errors());
547:                 }
548: 
549:                 $this->statement = $stmt;
550:                 $this->parameter_signature = $signature;
551:                     } else {
552:                 foreach ($converted as $index => $value) {
553:                     $this->bound_values[$index] = $value;
554:                 }
555:             }
556: 
557:             if (!@sqlsrv_execute($this->statement)) {
558:                 throw $this->query_error($this->errors());
559:             }
560: 
561:             $this->executed = true;
562: 
563:             // Send all stream parameters before reading row counts or returned identity values.
564:             if ($temporary) {
565:                 while (@sqlsrv_send_stream_data($this->statement)) {
566:                 }
567: 
568:                 if (sqlsrv_errors(SQLSRV_ERR_ERRORS)) {
569:                     throw $this->stream_error($this->errors());
570:                 }
571:             }
572: 
573:             $this->accept_statement();
574:         } catch (\Throwable $error) {
575:             $this->free_result();
576: 
577:             throw $error;
578:         } finally {
579:             foreach ($temporary as $stream) {
580:                 fclose($stream);
581:             }
582:         }
583:     } // native_execute

585:     /**
586:      * Captures SQL Server row counts, identity values and result metadata.
587:      *
588:      * Consumes leading row-count results and complete result-free batches. An appended
589:      * SCOPE_IDENTITY result is consumed internally. The first ordinary row result remains
590:      * available for public field access; its cursor is released when reading reaches EOF.
591:      *
592:      * @return void
593:      * Does not return a value.
594:      *
595:      * @throws DBWorkerException
596:      * If native result processing fails.
597:      *
598:      * @author Oleg Schildt
599:      */
600:     private function accept_statement(): void
601:     {
602:         if (($this->options['save_records_explicit_identity'] ?? false) && !$this->captures_identity) {
603:             // Execute every part of an explicit-ID batch, including its cleanup and deferred error results.
604:             $affected = 0;
605: 
606:             do {
607:                 $count = sqlsrv_rows_affected($this->statement);
608: 
609:                 if ($count !== false && $count >= 0) {
610:                     $affected = $count;
611:                 }
612: 
613:                 $next = sqlsrv_next_result($this->statement);
614: 
615:                 if ($next === false) {
616:                     throw $this->query_error($this->errors());
617:                 }
618:             } while ($next === true);
619: 
620:             $this->executed = false;
621: 
622:             $this->set_result([], $affected);
623: 
624:             return;
625:         }
626: 
627:         // Treat DML counts separately from SELECT results and internal identity result sets.
628:         $affected = 0;
629: 
630:         if ($this->captures_identity) {
631:             // SCOPE_IDENTITY must execute in the same batch as INSERT.
632:             $identity_result = false;
633: 
634:             do {
635:                 if (sqlsrv_num_fields($this->statement) === 0) {
636:                     $count = sqlsrv_rows_affected($this->statement);
637: 
638:                     if ($count !== false && $count >= 0) {
639:                         $affected = $count;
640:                     }
641:                 }
642: 
643:                 $fields = sqlsrv_field_metadata($this->statement);
644: 
645:                 if ($fields && $fields[0]['Name'] === 'smartfactory_insert_id') {
646:                     $identity_result = true;
647:                     $status = sqlsrv_fetch($this->statement);
648: 
649:                     if ($status === false) {
650:                         throw $this->query_error($this->errors());
651:                     }
652: 
653:                     if ($status === true) {
654:                         $id = sqlsrv_get_field($this->statement, 0);
655: 
656:                         if ($id !== null && filter_var($id, FILTER_VALIDATE_INT) !== false) {
657:                             $this->last_insert_id = (int) $id;
658:                         }
659:                     }
660: 
661:                     break;
662:                 }
663: 
664:                 $next = sqlsrv_next_result($this->statement);
665: 
666:                 if ($next === false) {
667:                     throw $this->query_error($this->errors());
668:                 }
669:             } while ($next === true);
670: 
671:             // Fetching the identity value alone leaves its result pending when MARS is disabled.
672:             if ($identity_result) {
673:                 $this->finish_results();
674:             } else {
675:                 $this->executed = false;
676:             }
677: 
678:             $this->set_result([], $affected);
679: 
680:             return;
681:         }
682: 
683:         while (true) {
684:             $field_count = sqlsrv_num_fields($this->statement);
685: 
686:             if ($field_count === false) {
687:                 throw $this->query_error($this->errors());
688:             }
689: 
690:             if ($field_count > 0) {
691:                 break;
692:             }

694:             $count = sqlsrv_rows_affected($this->statement);

696:             if ($count === false) {
697:                 throw $this->query_error($this->errors());
698:             }

702:             }

707:                 throw $this->query_error($this->errors());
708:             }

710:             if ($next === null) {
711:                 $this->executed = false;
712: 
713:                 $this->set_result([], $affected);
714: 
715:                 return;
716:             }
717:         }
718: 
719:         $fields = sqlsrv_field_metadata($this->statement);
720: 
721:         if ($fields === false && sqlsrv_errors(SQLSRV_ERR_ERRORS)) {
722:             throw $this->query_error($this->errors());
723:         }
724: 
725:         $types = [
726:             -5 => 'bigint',
727:             -2 => 'binary',
728:             -7 => 'bit',
729:             1 => 'char',
730:             91 => 'date',
731:             93 => 'datetime',
732:             -155 => 'datetimeoffset',
733:             3 => 'decimal',
734:             6 => 'float',
735:             -4 => 'image',
736:             4 => 'int',
737:             -8 => 'nchar',
738:             -10 => 'ntext',
739:             2 => 'numeric',
740:             -9 => 'nvarchar',
741:             7 => 'real',
742:             5 => 'smallint',
743:             -1 => 'text',

749:             12 => 'varchar',

757:             $numeric = in_array($type, [-5, -7, 3, 6, 4, 2, 7, 5, -6], true);
758:             $datetime = in_array($type, [91, 93, -155, -154], true);
759:             $metadata[] = [
760:                 'name' => $field['Name'],
761:                 'type' => $types[$type] ?? 'undefined',
762:                 'size' => $field['Size'] ?? 0,

764:                 'scale' => $field['Scale'] ?? null,

766:                 'binary' => (int) $binary,

768:                 'string' => (int) in_array($type, [1, -8, -10, -9, -1, 12, -152], true)
769:             ];

820:     /**
821:      * Fetches the next row through the native driver.
822:      *
823:      * @return list<mixed>|null
824:      * Returns a numerically indexed row, or NULL at the end of the result set.
825:      *
826:      * @throws DBWorkerException
827:      * If fetching or spooling a field fails, or a trailing batch result reports a deferred SQL error.
828:      *
829:      * @see DBWorker::fetch_row()
830:      *
831:      * @author Oleg Schildt
832:      */
833:     protected function native_fetch(): ?array
834:     {
835:         if (!$this->executed) {
836:             return null;
837:         }
838: 
839:         // Preserve numeric field positions and distinguish end-of-result from a native error.
840:         $status = sqlsrv_fetch($this->statement);
841: 
842:         if ($status === false) {
843:             $error = $this->query_error($this->errors());
844: 
845:             $this->free_result();
846: 
847:             throw $error;
848:         }
849: 
850:         if ($status === null) {
851:             try {
852:                 $this->finish_results();
853:             } catch (\Throwable $error) {
854:                 $this->free_result();
855: 
856:                 throw $error;
857:             }
858: 
859:             return null;
860:         }
861: 
862:         $row = [];
863: 
864:         try {
865:             foreach ($this->metadata as $index => $field) {
866:                 $large = ($field['binary'] || $field['string']) && ($field['size'] <= 0 || $field['size'] > 8192);
867: 
868:                 if ($large) {
869:                     $unicode = in_array($field['type'], ['nvarchar', 'nchar', 'ntext'], true);
870:                     $value = sqlsrv_get_field(
871:                         $this->statement,
872:                         $index,
873:                         SQLSRV_PHPTYPE_STREAM($field['binary'] || $unicode ? SQLSRV_ENC_BINARY : 'UTF-8')
874:                     );
875: 
876:                     if ($value === false) {
877:                         throw $this->query_error($this->errors());
878:                     }
879: 
880:                     if ($value !== null) {
881:                         $spool = tmpfile();
882: 
883:                         if ($spool === false) {
884:                             fclose($value);
885: 
886:                             throw $this->stream_error('Cannot spool a SQL Server result field.');
887:                         }
888: 
889:                         // Own the spool before copying so the failure path can always close it.
890:                         $row[$index] = $spool;
891: 
892:                         try {
893:                             // The sqlsrv UTF-8 stream converter can split a UTF-16 surrogate
894:                             // pair at its internal buffer boundary. Read Unicode fields as
895:                             // raw UTF-16LE and convert complete code points ourselves.
896:                             $carry = '';
897: 
898:                             $this->write_chunks(
899:                                 $value,
900:                                 function (string $chunk) use ($spool, $unicode, &$carry): void {
901:                                     if ($unicode) {
902:                                         $chunk = $carry . $chunk;
903:                                         $even = strlen($chunk) - strlen($chunk) % 2;
904:                                         $carry = substr($chunk, $even);
905:                                         $chunk = substr($chunk, 0, $even);
906: 
907:                                         if ($chunk !== '') {
908:                                             $last = unpack('v', substr($chunk, -2))[1];
909: 
910:                                             if ($last >= 0xD800 && $last <= 0xDBFF) {
911:                                                 $carry = substr($chunk, -2) . $carry;
912:                                                 $chunk = substr($chunk, 0, -2);
913:                                             }
914:                                         }
915: 
916:                                         if (!mb_check_encoding($chunk, 'UTF-16LE')) {
917:                                             throw $this->stream_error('Invalid UTF-16 result data.');
918:                                         }
919: 
920:                                         $chunk = mb_convert_encoding($chunk, 'UTF-8', 'UTF-16LE');
921:                                     }
922: 
923:                                     $this->write_all($spool, $chunk);
924:                                 }
925:                             );
926: 
927:                             if ($carry !== '') {
928:                                 throw $this->stream_error('Incomplete UTF-16 result data.');
929:                             }
930:                         } finally {
931:                             fclose($value);
932:                         }
933: 
934:                         rewind($spool);
935: 
936:                         continue;
937:                     }
938:                 } else {
939:                     $value = sqlsrv_get_field($this->statement, $index);
940: 
941:                     if ($value === false && sqlsrv_errors(SQLSRV_ERR_ERRORS)) {
942:                         throw $this->query_error($this->errors());
943:                     }
944:                 }
945: 
946:                 $row[$index] = $value;
947:             }
948: 
949:             return $row;
950:         } catch (\Throwable $error) {
951:             // Release any already spooled fields if a later field cannot be read.
952:             foreach ($row as $value) {
953:                 if (is_resource($value)) {
954:                     fclose($value);
955:                 }
956:             }
957: 
958:             $this->free_result();
959: 
960:             throw $error;
961:         }
962:     } // native_fetch

1009:     /**
1010:      * Performs a native transaction operation on the shared connection.
1011:      *
1012:      * The base class updates the shared transaction flag only after this operation succeeds.
1013:      *
1014:      * @param string $action
1015:      * The operation: begin, commit or rollback.
1016:      *
1017:      * @param string $isolation_level
1018:      * The validated isolation level for begin; ignored for commit and rollback.
1019:      * The previous session level is restored after the transaction ends.
1020:      *
1021:      * @return void
1022:      * Does not return a value.
1023:      *
1024:      * @throws DBWorkerException
1025:      * If reading or applying session isolation, starting, committing or rolling back fails.
1026:      *
1027:      * @author Oleg Schildt
1028:      */
1029:     protected function native_transaction(string $action, string $isolation_level = self::ISOLATION_READ_COMMITTED): void
1030:     {
1031:         // Perform the native operation; the base worker updates the shared flag after success.
1032:         if ($action === 'begin') {
1033:             $previous = $this->session_transaction_isolation();
1034: 
1035:             $this->session_transaction_isolation($isolation_level);
1036: 
1037:             if (!sqlsrv_begin_transaction($this->connection_state->handle)) {
1038:                 $error = $this->query_error($this->errors());
1039: 
1040:                 $this->session_transaction_isolation($previous);
1041: 
1042:                 throw $error;
1043:             }
1044: 
1045:             $this->connection_state->attributes['previous_transaction_isolation'] = $previous;
1046: 
1047:             return;
1048:         }
1049: 
1050:         $success = match ($action) {
1051:             'commit' => sqlsrv_commit($this->connection_state->handle),
1052:             'rollback' => sqlsrv_rollback($this->connection_state->handle)
1053:         };
1054: 
1055:         if (!$success) {
1056:             throw $this->query_error($this->errors());
1057:         }
1058:     } // native_transaction

1060:     /**
1061:      * Reads or sets SQL Server session isolation without replacing the worker's public statement.
1062:      *
1063:      * @param string|null $isolation_level
1064:      * A validated native isolation name to apply, or NULL to read the current level.
1065:      *
1066:      * @return string
1067:      * Returns the applied or current native isolation level.
1068:      *
1069:      * @throws DBWorkerException
1070:      * If SQL Server cannot execute the session command or return its isolation setting.
1071:      *
1072:      * @author Oleg Schildt
1073:      */
1074:     private function session_transaction_isolation(?string $isolation_level = null): string
1075:     {
1076:         // Execute SET directly: a setting made inside a prepared RPC can end with that RPC's scope.
1077:         $sql = $isolation_level === null
1078:             ? 'SELECT transaction_isolation_level FROM sys.dm_exec_sessions WHERE session_id = @@SPID'
1079:             : 'SET TRANSACTION ISOLATION LEVEL ' . $isolation_level;

1081: 
1082:         if ($statement === false) {
1083:             throw $this->query_error($this->errors());
1084:         }
1085: 
1086:         try {
1087:             if ($isolation_level === null) {
1088:                 $row = sqlsrv_fetch_array($statement, SQLSRV_FETCH_NUMERIC);
1089:

1091:                     throw $this->query_error('SQL Server did not return the session isolation level.');
1092:                 }
1093: 
1094:                 // Restore legacy session defaults as well, even though start_transaction() accepts only three levels.
1095:                 $levels = [1 => 'READ UNCOMMITTED', 2 => self::ISOLATION_READ_COMMITTED,
1096:                     3 => self::ISOLATION_REPEATABLE_READ, 4 => self::ISOLATION_SERIALIZABLE, 5 => 'SNAPSHOT'];
1097:                 $isolation_level = $levels[(int) $row[0]] ?? self::ISOLATION_READ_COMMITTED;
1098:             }
1099: 
1100:             return $isolation_level;
1101:         } finally {
1102:             sqlsrv_free_stmt($statement);
1103:         }
1104:     } // session_transaction_isolation

1215:     /**
1216:      * Resolves SQL Server table identity, column types and collation-aware comparison fingerprints.
1217:      *
1218:      * @param string $table
1219:      * The unquoted table, schema.table or database.schema.table name.
1220:      *
1221:      * @return array
1222:      * Returns canonical table names and native key metadata.
1223:      *
1224:      * @throws DBWorkerException
1225:      * If the target cannot be resolved or its column metadata cannot be read.
1226:      *
1227:      * @author Oleg Schildt
1228:      */
1229:     protected function native_save_table(string $table): array
1230:     {
1231:         $parts = explode('.', $table);
1232:         $name = array_pop($parts);
1233:         $schema = array_pop($parts);
1234:         $database = array_pop($parts) ?? $this->save_sql('SELECT DB_NAME()')['rows'][0][0];
1235:         $database = $this->save_sql('SELECT DB_NAME(DB_ID(?))', [$database])['rows'][0][0];
1236:         $quote = static fn (string $name): string => '[' . str_replace(']', ']]', $name) . ']';
1237:         $prefix = $quote($database);
1238:         $lookup = $prefix . '.' . ($schema === null ? '' : $quote($schema)) . '.' . $quote($name);
1239:         $result = $this->save_sql('SELECT s.name, t.name FROM ' . $prefix . '.sys.tables t JOIN ' . $prefix
1240:             . '.sys.schemas s ON s.schema_id = t.schema_id WHERE t.object_id = OBJECT_ID(?)', [$lookup]);
1241: 
1242:         if (count($result['rows']) !== 1) {
1243:             throw $this->query_error('The save_records table was not found.');
1244:         }
1245: 
1246:         [$schema, $name] = $result['rows'][0];
1247:         $qualified = $prefix . '.' . $quote($schema) . '.' . $quote($name);
1248:         $rows = $this->save_sql('SELECT c.name, t.name, c.max_length, c.precision, c.scale, c.collation_name, c.is_identity '
1249:             . 'FROM ' . $prefix . '.sys.columns c JOIN ' . $prefix . '.sys.types t ON t.user_type_id = c.system_type_id '
1250:             . 'WHERE c.object_id = OBJECT_ID(?)', [$qualified]);
1251:         $context = ['table' => $qualified, 'lock_table' => $prefix . '.[dbo].[dbworker_locks]',
1252:             'resource' => [$database, $schema, $name], 'columns' => []];
1253: 
1254:         foreach ($rows['rows'] as [$field, $type, $length, $precision, $scale, $collation, $identity]) {
1255:             $native = match ($type) {
1256:                 'varchar', 'char', 'binary', 'varbinary' => $type . '(' . ($length < 0 ? 'max' : $length) . ')',
1257:                 'nvarchar', 'nchar' => $type . '(' . ($length < 0 ? 'max' : (int) $length / 2) . ')',
1258:                 'decimal', 'numeric' => $type . '(' . $precision . ',' . $scale . ')',
1259:                 'datetime2', 'datetimeoffset', 'time' => $type . '(' . $scale . ')',
1260:                 default => $type
1261:             };
1262:             $cast = 'CAST(? AS ' . $native . ')' . ($collation === null ? '' : ' COLLATE ' . $collation);
1263:             $supported = in_array($type, ['varchar', 'char', 'nvarchar', 'nchar', 'binary', 'varbinary', 'decimal', 'numeric',
1264:                 'bigint', 'int', 'smallint', 'tinyint', 'bit', 'float', 'real', 'money', 'smallmoney', 'date', 'datetime',
1265:                 'smalldatetime', 'datetime2', 'datetimeoffset', 'time', 'uniqueidentifier'], true);
1266:             // Preserve full numeric entropy; use CHECKSUM only where SQL collation/equality requires it.
1267:             $fingerprint = 'CHECKSUM(' . $cast . ')';
1268: 
1269:             if (in_array($type, ['bigint', 'int', 'smallint', 'tinyint', 'bit', 'decimal', 'numeric'], true)) {
1270:                 $fingerprint = 'CONVERT(varchar(100), ' . $cast . ')';
1271:             } elseif ($collation !== null && preg_match('/_BIN2?\z/i', $collation)) {
1272:                 $fingerprint = 'CONVERT(varchar(max), CONVERT(varbinary(max), RTRIM(' . $cast . ')), 2)';
1273:             }
1274: 
1275:             $context['columns'][strtolower($field)] = ['name' => $field, 'cast' => $cast,
1276:                 'fingerprint' => $supported ? $fingerprint : null, 'identity' => (bool) $identity];
1277:         }
1278: 
1279:         return $context;
1280:     } // native_save_table
