12: /**
13:  * Binds implementations to interfaces or parent classes and creates initialized objects.
14:  *
15:  * Bindings can also map a class to itself, allowing a derived implementation to be
16:  * substituted later without changing business logic. Singleton instances are shared
17:  * per context and requested contract, rather than per implementation class.
18:  * Contracts are specified by class/interface names using ::class. Creation functions
19:  * declare a compatible, non-nullable class or interface as their native return type.
20:  * Each binding requires a named function or closure that accepts no arguments and
21:  * constructs, initializes and returns an implementation. The factory never invokes
22:  * constructors directly and validates each result before returning or caching it.
23:  *
24:  * @author Oleg Schildt
25:  */

64:     /**
65:      * Binds a factory function to an interface or class in the specified context.
66:      *
67:      * Contexts have independent bindings and singleton instances, with no fallback to
68:      * the default context. Specify the requested contract using ::class. Omitting
69:      * the context registers the default binding. The creation function's declared
70:      * return type is checked against the contract without invoking the function.
71:      * Existing bindings can be replaced before a singleton is created. Reset that
72:      * singleton explicitly before replacing its creation function.
73:      *
74:      * Examples:
75:      *
76:      *     ObjectFactory::bindFactory(ILanguageManager::class, function (): LanguageManager {
77:      *         $instance = new LanguageManager();
78:      *
79:      *         $instance->detectLanguage();
80:      *
81:      *         return $instance;
82:      *     });
83:      *
84:      *     ObjectFactory::bindFactory(IRecordsetManager::class, function (): RecordsetManager {
85:      *         $instance = new RecordsetManager();
86:      *
87:      *         $instance->setDBWorker(singleton(\SmartFactory\DatabaseWorkers\MySQL_DBWorker::class));
88:      *
89:      *         return $instance;
90:      *     }, "console");
91:      *
92:      * @template T of object
93:      *
94:      * @param class-string<T> $interface_or_class
95:      * The interface or class name of the requested contract, supplied using ::class.
96:      *
97:      * @param callable(): T $factory_function
98:      * A named function or closure called without arguments when an instance is requested.
99:      * It must declare one non-nullable class or interface return type compatible with
100:      * the contract and return a fully initialized implementation on each call.
101:      * Missing, built-in, union and intersection return types are rejected.
102:      * Method callbacks and invokable objects are not supported.
103:      *
104:      * @param string $context
105:      * The binding context; defaults to "default". Any string, including an empty string, is accepted.
106:      *
107:      * @return void
108:      * Registers the binding without creating an instance.
109:      *
110:      * @throws ObjectFactoryException
111:      * If the contract or declared return type is unknown, the return declaration is
112:      * missing, unsupported, nullable or incompatible with the contract, the creation
113:      * function is unsupported or requires arguments, or the contract has a cached
114:      * singleton or is currently being resolved.
115:      *
116:      * @throws \Throwable
117:      * If autoloading the contract or declared return type throws an exception or error, it is propagated.
118:      *
119:      * @author Oleg Schildt
120:      */
121:     public static function bindFactory(string $interface_or_class, callable $factory_function, string $context = "default"): void
122:     {
123:         $contract = self::reflectClass($interface_or_class, $context);
124:         $contract_name = $contract->getName();
125:         $description = sprintf("context '%s', contract '%s'", $context, $contract_name);
126: 
127:         self::assertMutable($context, $contract_name);
128: 
129:         if (isset(self::$singletons[$context][$contract_name])) {
130:             throw new ObjectFactoryException(sprintf("Cannot replace binding (%s): reset the singleton first.", $description));
131:         }
132: 
133:         if (!$factory_function instanceof \Closure && (!is_string($factory_function) || !function_exists($factory_function))) {
134:             throw new ObjectFactoryException(sprintf("Cannot register binding (%s): the creation function must be a named function or closure.", $description));
135:         }
136: 
137:         $creator = new \ReflectionFunction($factory_function);
138: 
139:         if ($creator->getNumberOfRequiredParameters() > 0) {
140:             throw new ObjectFactoryException(sprintf("Cannot register binding (%s): the creation function must be callable without arguments.", $description));
141:         }
142: 
143:         $return_class = self::reflectReturnClass($creator, $context, $contract_name);
144:         $return_class_name = $return_class->getName();
145: 
146:         if (!is_a($return_class_name, $contract_name, true)) {
147:             throw new ObjectFactoryException(sprintf(
148:                 "Cannot register binding (%s, declared return type '%s'): the return type must implement or extend the contract.",
149:                 $description,
150:                 $return_class_name
151:             ));
152:         }
153: 
154:         self::$itable[$context][$contract_name] = ["return_type" => $return_class_name, "factory_function" => $creator];
155:     } // bindFactory

157:     /**
158:      * Creates and initializes an implementation of the requested contract.
159:      *
160:      * Calls the registered creation function without arguments and verifies that its
161:      * result implements or extends the requested contract.
162:      * A singleton is cached only after successful creation and result validation.
163:      * Failed requests can be retried. Recursive requests for the same contract in
164:      * the same context are rejected throughout the creation function's execution.
165:      * Singleton identity is scoped to the context and contract; different contracts
166:      * bound to the same implementation can have different singleton instances.
167:      *
168:      * @template T of object
169:      *
170:      * @param class-string<T> $interface_or_class
171:      * The interface or class name of the requested contract, supplied using ::class.
172:      *
173:      * @param bool $singleton
174:      * Whether to reuse a cached instance for this context and contract. If false, calls the creation function every time.
175:      *
176:      * @param string $context
177:      * The exact binding context, without fallback to the default context.
178:      *
179:      * @return T
180:      * Returns the fully initialized implementation of the contract.
181:      *
182:      * @throws ObjectFactoryException
183:      * If the contract name is missing or unknown, no binding exists in the context,
184:      * a circular dependency is detected, or the creation function returns a value
185:      * that does not implement or extend the requested contract.
186:      *
187:      * @throws \Throwable
188:      * If autoloading or the creation function throws an exception or error, it is
189:      * propagated unchanged and the failed instance is not cached.
190:      *
191:      * @used_by \SmartFactory\instance()
192:      * @used_by \SmartFactory\singleton()
193:      *
194:      * @author Oleg Schildt
195:      */
196:     public static function getInstance(string $interface_or_class, bool $singleton, string $context = "default"): object
197:     {
198:         $contract = self::reflectClass($interface_or_class, $context);
199:         $contract_name = $contract->getName();
200: 
201:         if (!isset(self::$itable[$context][$contract_name])) {
202:             throw new ObjectFactoryException(sprintf("No binding exists for contract '%s' in context '%s'.", $contract_name, $context));
203:         }
204: 
205:         $binding = self::$itable[$context][$contract_name];
206:         $description = sprintf("[context '%s', contract '%s', declared return type '%s']", $context, $contract_name, $binding["return_type"]);
207: 
208:         if (isset(self::$resolving[$context][$contract_name])) {
209:             throw new ObjectFactoryException("Circular factory dependency: " . implode(" -> ", [...self::$resolution_path, $description]));
210:         }
211: 
212:         if ($singleton && isset(self::$singletons[$context][$contract_name])) {
213:             return self::$singletons[$context][$contract_name];
214:         }
215: 
216:         self::$resolving[$context][$contract_name] = true;
217:         self::$resolution_path[] = $description;
218: 
219:         try {
220:             $instance = $binding["factory_function"]->invoke();
221: 
222:             if (!is_object($instance) || !$contract->isInstance($instance)) {
223:                 throw new ObjectFactoryException(sprintf(
224:                     "Invalid creation result (%s): expected an instance of '%s', got '%s'.",
225:                     $description,
226:                     $contract_name,
227:                     get_debug_type($instance)
228:                 ));
229:             }
230: 
231:             // Publish only validated results; finally also cleans up failed requests.
232:             if ($singleton) {
233:                 self::$singletons[$context][$contract_name] = $instance;
234:             }
235: 
236:             return $instance;
237:         } finally {
238:             unset(self::$resolving[$context][$contract_name]);
239: 
240:             if (self::$resolving[$context] === []) {
241:                 unset(self::$resolving[$context]);
242:             }
243: 
244:             array_pop(self::$resolution_path);
245:         }
246:     } // getInstance

342:     /**
343:      * Resolves a class/interface name to its canonical reflection.
344:      *
345:      * @param class-string $class
346:      * An existing class/interface name, including aliases and alternate casing.
347:      *
348:      * @param string $context
349:      * The operation's context, included in validation errors.
350:      *
351:      * @return \ReflectionClass<object>
352:      * Returns reflection with the canonical class or interface name.
353:      *
354:      * @throws ObjectFactoryException
355:      * If the name is empty or does not identify an existing class or interface.

371:     /**
372:      * Resolves the creation function's native return declaration to a class or interface.
373:      *
374:      * Only one non-nullable named object type is accepted. Contextual names self,
375:      * parent and static are resolved using the closure's lexical or called class.
376:      * The creation function is not invoked.
377:      *
378:      * @param \ReflectionFunction $creator
379:      * The registered named function or closure to inspect.
380:      *
381:      * @param string $context
382:      * The binding context, included in validation errors.
383:      *
384:      * @param class-string $contract_name
385:      * The canonical requested contract name, included in validation errors.
386:      *
387:      * @return \ReflectionClass<object>
388:      * Returns reflection for the existing class or interface declared as the return type.
389:      *
390:      * @throws ObjectFactoryException
391:      * If the return declaration is missing, built-in, nullable, a union or intersection,
392:      * names an unknown type, or contains a contextual name without a matching class scope.
393:      *
394:      * @throws \Throwable
395:      * If autoloading the declared return type throws an exception or error, it is propagated.
396:      *
397:      * @author Oleg Schildt
398:      */
399:     protected static function reflectReturnClass(\ReflectionFunction $creator, string $context, string $contract_name): \ReflectionClass
400:     {
401:         $return_type = $creator->getReturnType();
402:         $description = sprintf("context '%s', contract '%s'", $context, $contract_name);
403: 
404:         if (!$return_type instanceof \ReflectionNamedType || $return_type->isBuiltin() || $return_type->allowsNull()) {
405:             throw new ObjectFactoryException(sprintf(
406:                 "Cannot register binding (%s): the creation function must declare one non-nullable class or interface return type; got '%s'.",
407:                 $description,
408:                 $return_type === null ? "no return type" : (string) $return_type
409:             ));
410:         }
411: 
412:         $return_name = $return_type->getName();
413: 
414:         // Reflection preserves contextual names instead of expanding them to class names.
415:         if (in_array($return_name, ["self", "parent", "static"], true)) {
416:             $scope = $creator->getClosureScopeClass();
417:             $return_class = match ($return_name) {
418:                 "self" => $scope,
419:                 "parent" => $scope === null ? false : $scope->getParentClass(),
420:                 "static" => $creator->getClosureCalledClass()
421:             };
422: 
423:             if (!$return_class instanceof \ReflectionClass) {
424:                 throw new ObjectFactoryException(sprintf(
425:                     "Cannot register binding (%s): cannot resolve declared return type '%s' without a matching class scope.",
426:                     $description,
427:                     $return_name
428:                 ));
429:             }
430: 
431:             return $return_class;
432:         }
433: 
434:         if (!interface_exists($return_name) && !class_exists($return_name)) {
435:             throw new ObjectFactoryException(sprintf(
436:                 "Cannot register binding (%s): declared return type '%s' does not exist.",
437:                 $description,
438:                 $return_name
439:             ));
440:         }
441: 
442:         return new \ReflectionClass($return_name);
443:     } // reflectReturnClass

451:      * @param ?string $contract_name
452:      * The canonical contract name, or null to check the entire selected context.
453:      *
454:      * @return void
