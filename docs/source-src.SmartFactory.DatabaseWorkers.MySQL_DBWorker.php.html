14:  * Database worker for MySQL using the mysqli extension.
15:  *
16:  * Implements native query execution, prepared binding and chunked LOB uploads.

149:     /**
150:      * Establishes the database connection using the initialized settings.
151:      *
152:      * Uses utf8mb4 and caches the current NO_BACKSLASH_ESCAPES mode.
153:      * An optional isolation_level or read_only setting is applied to the session.
154:      * An already connected worker keeps its existing connection.
155:      *
156:      * @return void
157:      * Does not return a value.
158:      *
159:      * @throws DBWorkerException
160:      * If connection settings are incomplete, the isolation level is invalid, or connection or session
161:      * initialization fails.
162:      *
163:      * @see DBWorker::init()
164:      * @see MySQL_DBWorker::is_connected()
165:      * @see DBWorker::close_connection()
166:      *
167:      * @author Oleg Schildt
168:      */
169:     public function connect(): void
170:     {
171:         // Reuse an attached connection and validate the settings before creating a new native handle.
172:         if ($this->is_connected()) {
173:             return;
174:         }
175: 
176:         if ($this->db_server === '' || $this->db_user === '') {
177:             throw new DBWorkerException('Connection data is incomplete.', self::ERR_CONNECTION_DATA_INCOMPLETE);
178:         }
179: 
180:         mysqli_report(MYSQLI_REPORT_ERROR | MYSQLI_REPORT_STRICT);
181: 
182:         $connection = mysqli_init();
183: 
184:         try {
185:             $connection->options(MYSQLI_OPT_CONNECT_TIMEOUT, (int) ($this->options['connect_timeout'] ?? 20));
186: 
187:             $connection->options(MYSQLI_OPT_READ_TIMEOUT, (int) ($this->options['read_timeout'] ?? 20));
188: 
189:             $connection->real_connect(
190:                 $this->db_server,
191:                 $this->db_user,
192:                 $this->db_password,
193:                 $this->db_name === '' ? null : $this->db_name,
194:                 $this->db_port === '' ? null : (int) $this->db_port
195:             );
196: 
197:             // Configure the native connection encoding before any escaping or application query.
198:             $connection->set_charset('utf8mb4');
199: 
200:             $this->connection_state = new DBConnectionState($connection, static fn (\mysqli $db) => $db->close());
201: 
202:             $this->refresh_sql_mode();
203: 
204:             if (isset($this->options['isolation_level'])) {
205:                 $level = strtoupper((string) $this->options['isolation_level']);
206: 
207:                 if (
208:                     !in_array($level, ['READ UNCOMMITTED', 'READ COMMITTED', 'REPEATABLE READ', 'SERIALIZABLE'], true)
209:                 ) {
210:                     throw new DBWorkerException('Invalid transaction isolation level.', self::ERR_WRONG_DATA_FORMAT);
211:                 }
212: 
213:                 $connection->query('SET SESSION TRANSACTION ISOLATION LEVEL ' . $level);
214:             }
215: 
216:             if ($this->read_only) {
217:                 $connection->query('SET SESSION TRANSACTION READ ONLY');
218:             }
219:         } catch (\Throwable $error) {
220:             $this->connection_state = null;
221: 
222:             try {
223:                 $connection->close();
224:             } catch (\Throwable) {
225:             }
226: 
227:             if ($error instanceof DBWorkerException) {
228:                 throw $error;
229:             }
230: 
231:             $code = match ($error->getCode()) {
232:                 2002, 2003 => self::ERR_HOST_UNREACHABLE,
233:                 1045 => self::ERR_WRONG_USER_CREDENTIALS,
234:                 1049 => self::ERR_DATABASE_NOT_FOUND,
235:                 default => self::ERR_CONNECTION_FAILED
236:             };
237: 
238:             throw new DBWorkerException($error->getMessage(), $code);
239:         }

341:     /**
342:      * Binds validated values and executes the native prepared statement.
343:      *
344:      * Scalar PHP types determine mysqli bind types. LOBs are sent in chunks through
345:      * send_long_data(); empty LOBs remain distinct from NULL. Result rows are read
346:      * without buffering the result set, and bound-row references remain stable.
347:      *
348:      * @param list<scalar|null|BlobValue|ClobValue> $values
349:      * The normalized values in SQL occurrence order, including any explicit LOB wrappers.
350:      *
351:      * @return void
352:      * Does not return a value.
353:      *
354:      * @throws DBWorkerException
355:      * If binding, LOB processing or native execution fails.
356:      *
357:      * @see DBWorker::execute_prepared_query()
358:      *
359:      * @author Oleg Schildt
360:      */
361:     protected function native_execute(array $values): void
362:     {
363:         // Bind stable per-position values so repeated application names cannot share a loop variable by accident.
364:         $types = '';
365:         $bound = [];
366:         $lobs = [];
367: 
368:         foreach ($values as $index => $value) {
369:             if ($value instanceof BlobValue || $value instanceof ClobValue) {
370:                 $types .= 'b';
371:                 $bound[$index] = null;
372:                 $lobs[$index] = $value;
373:             } elseif (is_int($value) || is_bool($value)) {
374:                 $types .= 'i';
375:                 $bound[$index] = (int) $value;
376:             } elseif (is_float($value)) {
377:                 if (!is_finite($value)) {
378:                     throw $this->query_error('Non-finite numeric parameter.');
379:                 }
380: 
381:                 $types .= 'd';
382:                 $bound[$index] = $value;
383:             } else {
384:                 $types .= 's';
385:                 $bound[$index] = $value;
386:             }
387:         }
388: 
389:         if ($values && !$this->statement->bind_param($types, ...$bound)) {
390:             throw $this->query_error($this->statement->error);
391:         }
392: 
393:         // Send each LOB to its own native position, preserving repeated named bindings.
394:         foreach ($lobs as $index => $lob) {
395:             $sent = false;
396: 
397:             $this->write_chunks(
398:                 $lob->value,
399:                 function (string $chunk) use ($index, &$sent): void {
400:                     if (!$this->statement->send_long_data($index, $chunk)) {
401:                         throw $this->query_error($this->statement->error);
402:                     }
403: 
404:                     $sent = true;
405:                 },
406:                 $lob instanceof ClobValue
407:             );
408: 
409:             // An empty LOB is an empty value, not SQL NULL.
410:             if (!$sent) {
411:                 $this->statement->send_long_data($index, '');
412:             }
413:         }
414: 
415:         if (!$this->statement->execute()) {
416:             throw $this->query_error($this->statement->error);
417:         }
418: 
419:         $metadata = [];
420: 
421:         // Bind result columns without buffering rows; successful DML has no result metadata.
422:         if ($this->statement->field_count > 0) {
423:             $result = $this->statement->result_metadata();
424:             $metadata = $this->describe($result);
425: 
426:             $result->free();
427: 
428:             $this->bound_row = array_fill(0, count($metadata), null);
429: 
430:             // bind_result() needs stable references, not a temporary unpacked copy of row values.
431:             $references = [];
432: 
433:             foreach ($this->bound_row as &$value) {
434:                 $references[] = &$value;
435:             }
436: 
437:             unset($value);
438: 
439:             if (!$this->statement->bind_result(...$references)) {
440:                 throw $this->query_error($this->statement->error);
441:             }
442:         }
443: 
444:         $this->set_result($metadata, $this->statement->affected_rows);
445: 
446:         if (!($this->options['save_records_explicit_identity'] ?? false)) {
447:             $this->save_insert_id($this->statement->insert_id);
448:         }
449: 
450:         if (preg_match('/\ASET\b[\s\S]*\bsql_mode\b/i', $this->sql_body($this->compiled_query->original))) {
451:             $this->refresh_sql_mode();
452:         }
453:     } // native_execute

548:     /**
549:      * Releases native result resources without discarding a reusable prepared statement.
550:      *
551:      * Implementations must distinguish a result handle from a statement handle when both refer to the same resource.
552:      *
553:      * @return void
554:      * Does not return a value.
555:      *
556:      * @see DBWorker::free_result()
557:      *
558:      * @author Oleg Schildt
559:      */
560:     protected function native_free_result(): void
561:     {
562:         // Release result ownership while leaving a reusable prepared statement available.
563:         if ($this->result instanceof \mysqli_result) {
564:             $this->result->free();
565:         }
566: 
567:         $this->result = null;
568: 
569:         if ($this->statement instanceof \mysqli_stmt) {
570:             $this->statement->free_result();
571:         }
572: 
573:         $this->bound_row = [];
574: 
575:         // CALL can leave additional protocol results pending.
576:         if ($this->connection_state !== null && $this->statement === null) {
577:             $db = $this->connection_state->handle;
578: 
579:             // Drain additional CALL results before the shared connection accepts another query.
580:             while ($db->more_results()) {
581:                 $db->next_result();
582: 
583:                 $extra = $db->use_result();
584: 
585:                 if ($extra instanceof \mysqli_result) {
586:                     $extra->free();
587:                 }
588:             }
589:         }
590: 
591:         if ($this->statement instanceof \mysqli_stmt) {
592:             while ($this->statement->more_results()) {
593:                 $this->statement->next_result();
594: 
595:                 $this->statement->free_result();
596:             }
597:         }
598:     } // native_free_result

620:     /**
621:      * Performs a native transaction operation on the shared connection.
622:      *
623:      * The base class updates the shared transaction flag only after this operation succeeds.
624:      *
625:      * @param string $action
626:      * The operation: begin, commit or rollback.
627:      *
628:      * @param string $isolation_level
629:      * The validated isolation level applied only to the transaction being started.
630:      * Ignored for commit and rollback.
631:      *
632:      * @return void
633:      * Does not return a value.
634:      *
635:      * @throws DBWorkerException
636:      * If the native transaction operation fails.
637:      *
638:      * @author Oleg Schildt
639:      */
640:     protected function native_transaction(string $action, string $isolation_level = self::ISOLATION_READ_COMMITTED): void
641:     {
642:         // Perform the native operation; the base worker updates the shared flag after success.
643:         $db = $this->connection_state->handle;
644: 
645:         try {
646:             if ($action === 'begin') {
647:                 // Without SESSION this applies to the next transaction and preserves the session default.
648:                 if ($db->query('SET TRANSACTION ISOLATION LEVEL ' . $isolation_level) === false) {
649:                     throw $this->query_error($db->error);
650:                 }
651:             }
652: 
653:             $success = match ($action) {
654:                 'begin' => $db->begin_transaction($this->read_only ? MYSQLI_TRANS_START_READ_ONLY : 0),
655:                 'commit' => $db->commit(),
656:                 'rollback' => $db->rollback()
657:             };
658: 
659:             if (!$success) {
660:                 throw $this->query_error($db->error);
661:             }
662:         } catch (\mysqli_sql_exception $error) {
663:             throw $this->query_error($error->getMessage());
664:         }
665:     } // native_transaction

793:     /**
794:      * Resolves a MySQL table and comparison expressions from information_schema.
795:      *
796:      * @param string $table
797:      * The unquoted table name, optionally prefixed by its database.
798:      *
799:      * @return array
800:      * Returns canonical table names and native key metadata.
801:      *
802:      * @throws DBWorkerException
803:      * If the target cannot be resolved or its column metadata cannot be read.
804:      *
805:      * @author Oleg Schildt
806:      */
807:     protected function native_save_table(string $table): array
808:     {
809:         $parts = explode('.', $table);
810: 
811:         if (count($parts) > 2) {
812:             throw new DBWorkerException('MySQL table names support database.table.', self::ERR_WRONG_DATA_FORMAT);
813:         }
814: 
815:         $name = array_pop($parts);
816:         $database = $parts[0] ?? $this->save_sql('SELECT DATABASE()')['rows'][0][0];
817:         $result = $this->save_sql('SELECT TABLE_SCHEMA, TABLE_NAME FROM information_schema.TABLES WHERE TABLE_SCHEMA = ? AND TABLE_NAME = ?', [$database, $name]);
818: 
819:         if (count($result['rows']) !== 1) {
820:             throw $this->query_error('The save_records table was not found.');
821:         }
822: 
823:         [$database, $name] = $result['rows'][0];
824:         $quote = static fn (string $name): string => '`' . str_replace('`', '``', $name) . '`';
825:         $columns = $this->save_sql(
826:             'SELECT c.COLUMN_NAME, c.DATA_TYPE, c.NUMERIC_PRECISION, c.NUMERIC_SCALE, c.CHARACTER_SET_NAME, '
827:             . 'c.COLLATION_NAME, co.PAD_ATTRIBUTE, c.EXTRA, c.CHARACTER_MAXIMUM_LENGTH, c.DATETIME_PRECISION '
828:             . 'FROM information_schema.COLUMNS c LEFT JOIN information_schema.COLLATIONS co ON co.COLLATION_NAME = c.COLLATION_NAME '
829:             . 'WHERE c.TABLE_SCHEMA = ? AND c.TABLE_NAME = ?', [$database, $name]
830:         );
831:         $context = ['table' => $quote($database) . '.' . $quote($name), 'lock_table' => $quote($database) . '.`dbworker_locks`',
832:             'resource' => [$database, $name], 'columns' => []];
833: 
834:         foreach ($columns['rows'] as $row) {
835:             [$field, $type, $precision, $scale, $charset, $collation, $pad, $extra, $length, $fraction] = $row;
836:             $cast = match ($type) {
837:                 'tinyint', 'smallint', 'mediumint', 'int', 'bigint', 'decimal', 'numeric' => 'CAST(? AS DECIMAL(65,' . (int) $scale . '))',
838:                 'float' => 'CAST(? AS FLOAT(24))',
839:                 'double', 'real' => 'CAST(? AS DOUBLE)',
840:                 'date' => 'CAST(? AS DATE)',
841:                 'datetime', 'timestamp' => 'CAST(? AS DATETIME(' . (int) $fraction . '))',
842:                 'time' => 'CAST(? AS TIME(' . (int) $fraction . '))',
843:                 default => '?'
844:             };
845:             $fingerprint = $cast;
846: 
847:             if (in_array($type, ['enum', 'set'], true)) {
848:                 $fingerprint = null;
849:             } elseif ($charset !== null) {
850:                 // Native sort weights preserve case/accent equivalence, including PAD SPACE collations.
851:                 $cast = 'CONVERT(? USING ' . $charset . ') COLLATE ' . $collation;
852:                 $source = $pad === 'PAD SPACE' || $type === 'char' ? 'RTRIM(' . $cast . ')' : $cast;
853:                 $fingerprint = 'HEX(WEIGHT_STRING(' . $source . '))';
854:             } elseif (in_array($type, ['binary', 'varbinary'], true)) {
855:                 $cast = $type === 'binary' ? 'CAST(? AS BINARY(' . (int) $length . '))' : 'CAST(? AS BINARY)';
856:                 $fingerprint = 'HEX(' . $cast . ')';
857:             } elseif (!in_array($type, ['tinyint', 'smallint', 'mediumint', 'int', 'bigint', 'decimal', 'numeric',
858:                 'float', 'double', 'real', 'date', 'datetime', 'timestamp', 'time'], true)) {
859:                 $fingerprint = null;
860:             }
861: 
862:             $context['columns'][strtolower($field)] = ['name' => $field, 'cast' => $cast, 'fingerprint' => $fingerprint,
863:                 'identity' => str_contains($extra, 'auto_increment'), 'length' => $length];
864:         }
865: 
866:         return $context;
867:     } // native_save_table

962:      * Preserves an explicitly supplied zero identity without changing the session's normal SQL mode.
963:      *
964:      * @param array $context
965:      * The resolved table and column metadata.
966:      *
967:      * @param array $values
968:      * The prepared field values, including an explicit identity when supplied.
969:      *
970:      * @param array $expressions
971:      * The positional SQL expression for each field.
972:      *
973:      * @param string $identity_field
974:      * The configured identity field, or an empty string.
975:      *
976:      * @return int|null
977:      * Returns the inserted identity, or NULL when none is configured.
978:      *
979:      * @throws DBWorkerException
980:      * If changing SQL mode, inserting the record or restoring the mode fails.
981:      *
982:      * @author Oleg Schildt
983:      */
984:     protected function native_save_insert(array $context, array $values, array $expressions, string $identity_field): ?int
985:     {
986:         if ($identity_field === '' || ($values[$identity_field] ?? null) !== 0) {
987:             return parent::native_save_insert($context, $values, $expressions, $identity_field);
988:         }
989: 
990:         $mode = $this->save_sql('SELECT @@SESSION.sql_mode')['rows'][0][0];
991: 
992:         if (in_array('NO_AUTO_VALUE_ON_ZERO', explode(',', $mode), true)) {
993:             return parent::native_save_insert($context, $values, $expressions, $identity_field);
994:         }
995: 
996:         // Zero is a known ID in this API, whereas MySQL normally interprets it as an allocation request.
997:         $this->save_sql('SET SESSION sql_mode = ?', [ltrim($mode . ',NO_AUTO_VALUE_ON_ZERO', ',')]);
998: 
999:         try {
1000:             return parent::native_save_insert($context, $values, $expressions, $identity_field);
1001:         } finally {
1002:             $this->save_sql('SET SESSION sql_mode = ?', [$mode]);
1003:         }
1004:     } // native_save_insert
