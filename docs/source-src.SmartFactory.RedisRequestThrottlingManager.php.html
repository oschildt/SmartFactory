192:     /**
193:      * Serializes a caller's admission and releases only the lock owned by this operation.
194:      *
195:      * @param string $key
196:      * The lowercase SHA-256 caller hash supplied by the common algorithm.
197:      *
198:      * @param callable(): int $operation
199:      * The clock/read/check/write operation invoked exactly once after lock acquisition.
200:      *
201:      * @return int
202:      * Returns admission or retry only after verifying ownership and releasing the lock.
203:      *
204:      * @throws \LogicException
205:      * If the client is unavailable, in queued mode, or admission is nested.
206:      *
207:      * @throws \RuntimeException
208:      * If Redis commands fail, lock contention exceeds the timeout, or the lease expires.
209:      * An expired owner cannot write history or release a replacement owner's lock.
210:      *
211:      * @throws \Throwable
212:      * If token generation or the callback fails. The original failure survives cleanup failures.
213:      * A write whose response is lost may consume a slot without granting admission.
214:      *
215:      * @author Oleg Schildt
216:      */
217:     protected function executeAtomically(string $key, callable $operation): int
218:     {
219:         if ($this->active_key !== null) {
220:             throw new \LogicException("Nested throttling admission operations are not supported.");
221:         }
222: 
223:         $redis = $this->client();
224:         [$lock] = $this->keys($key);
225:         $token = bin2hex(random_bytes(24));
226:         $started = hrtime(true);
227:         $failure = null;
228:         $acquired = false;
229:         $acquire_script = 'return redis.call("SET", KEYS[1], ARGV[1], "NX", "PX", ARGV[2]) and 1 or 0';
230:         $release_script = 'if redis.call("GET", KEYS[1]) == ARGV[1] then return redis.call("DEL", KEYS[1]) end return 0';
231: 
232:         try {
233:             do {
234:                 $result = $redis->eval($acquire_script, [$lock, $token, $this->lock_ttl_ms], 1);
235: 
236:                 if ($result === 1) {
237:                     $acquired = true;
238: 
239:                     break;
240:                 }
241: 
242:                 if ($result !== 0) {
243:                     throw new \RuntimeException("The Redis throttling lock could not be acquired.");
244:                 }
245: 
246:                 $remaining_ms = $this->lock_timeout_ms - (hrtime(true) - $started) / 1000000;
247: 
248:                 if ($remaining_ms <= 0) {
249:                     throw new \RuntimeException("The Redis throttling lock wait timed out.");
250:                 }
251: 
252:                 usleep((int) min($remaining_ms * 1000, $this->lock_retry_interval_ms * 1000));
253:             } while (true);
254: 
255:             $this->active_key = $key;
256:             $this->active_token = $token;
257: 
258:             return $operation();
259:         } catch (\RedisException $exception) {
260:             $failure = new \RuntimeException("The Redis throttling storage operation failed.", 0, $exception);
261: 
262:             throw $failure;
263:         } catch (\Throwable $exception) {
264:             $failure = $exception;
265: 
266:             throw $exception;
267:         } finally {
268:             $this->active_key = null;
269:             $this->active_token = null;
270: 
271:             try {
272:                 // Also attempt owner-checked release after an ambiguous acquisition response.
273:                 $released = $redis->eval($release_script, [$lock, $token], 1);
274: 
275:                 if ($acquired && $released !== 1 && $failure === null) {
276:                     throw new \RuntimeException("The Redis throttling lock expired before admission completed.");
277:                 }
278:             } catch (\Throwable $cleanup_exception) {
279:                 if ($failure === null) {
280:                     throw new \RuntimeException("The Redis throttling lock could not be released.", 0, $cleanup_exception);
281:                 }
282:             }
283:         }
284:     } // executeAtomically

338:      * Loads only the selected caller's ordered JSON history while checking lease ownership.
339:      *
340:      * @param string $key
341:      * The lowercase SHA-256 caller hash supplied to executeAtomically().
342:      *
343:      * @return list<float>
344:      * Returns finite, nonnegative, ascending timestamps, preserving duplicates. A missing key
345:      * returns an empty list. Existing empty, malformed, non-array or unordered history fails closed.
346:      *
347:      * @throws \LogicException
348:      * If access does not belong to the active caller or the client cannot execute commands.
349:      *
350:      * @throws \RuntimeException
351:      * If the lease was lost, a command reply is invalid, or the stored JSON history is invalid.
352:      *

354:      * If Redis cannot read the key; the atomic boundary wraps the storage failure.
355:      *
356:      * @author Oleg Schildt
357:      */
358:     protected function loadRecords(string $key): array
359:     {
360:         $token = $this->token($key);
361:         $script = 'if redis.call("GET", KEYS[1]) ~= ARGV[1] then return {0} end return {1, redis.call("GET", KEYS[2])}';
362:         $result = $this->client()->eval($script, [...$this->keys($key), $token], 2);
363: 
364:         if (!is_array($result) || count($result) !== 2 || $result[0] !== 1) {
365:             throw new \RuntimeException("The Redis throttling history could not be read under its owner lock.");
366:         }
367: 
368:         if ($result[1] === false || $result[1] === null) {
369:             return [];
370:         }
371: 
372:         if (!is_string($result[1]) || $result[1] === "") {
373:             throw new \RuntimeException("The stored Redis throttling history is empty or is not text.");
374:         }
375: 
376:         try {
377:             $values = json_decode($result[1], false, 512, JSON_THROW_ON_ERROR);
378:         } catch (\JsonException $exception) {
379:             throw new \RuntimeException("The stored Redis throttling history is invalid JSON.", 0, $exception);
380:         }
381: 
382:         if (!is_array($values)) {
383:             throw new \RuntimeException("The stored Redis throttling history must be a JSON array.");
384:         }
385: 
386:         $timestamps = [];
387:         $previous = -1.0;
388: 
389:         foreach ($values as $value) {
390:             if ((!is_int($value) && !is_float($value)) || !is_finite((float) $value) || $value < 0 || $value < $previous) {
391:                 throw new \RuntimeException("The stored Redis throttling timestamps are invalid or out of order.");
392:             }
393: 
394:             $timestamp = (float) $value;
395:             $timestamps[] = $timestamp;
396:             $previous = $timestamp;
397:         }
398: 
399:         return $timestamps;
400:     } // loadRecords

402:     /**
403:      * Replaces only the active caller's history if the lease is still owned, with atomic expiry.
404:      *
405:      * @param string $key
406:      * The lowercase SHA-256 caller hash supplied to executeAtomically().
407:      *
408:      * @param list<float> $timestamps
409:      * This caller's nonempty, ordered history supplied by the common algorithm.
410:      *
411:      * @return void
412:      * Stores JSON and a millisecond TTL covering the newest timestamp plus the longest window.
413:      * Extremely large retention beyond Redis/Lua's safe millisecond range is stored without expiry
414:      * to avoid discarding active history prematurely. Rejections never call this method or renew TTL.
415:      *
416:      * @throws \LogicException
417:      * If access is outside the active caller's operation or the timestamp list is empty.
418:      *
419:      * @throws \RuntimeException
420:      * If the lease was lost or Redis did not confirm persistence.
421:      *
422:      * @throws \JsonException
423:      * If the supplied history cannot be encoded as JSON.
424:      *
425:      * @throws \RedisException
426:      * If Redis cannot persist history; the atomic boundary wraps the storage failure.
427:      *
428:      * @author Oleg Schildt
429:      */
430:     protected function saveRecords(string $key, array $timestamps): void
431:     {
432:         $token = $this->token($key);
433: 
434:         if ($timestamps === []) {
435:             throw new \LogicException("An admitted throttling request must contain at least one timestamp.");
436:         }
437: 
438:         $json = json_encode($timestamps, JSON_THROW_ON_ERROR | JSON_PRESERVE_ZERO_FRACTION);
439:         $expires = max($timestamps) + max(array_column($this->limits, 1));
440:         $script = <<<'LUA'
441: if redis.call("GET", KEYS[1]) ~= ARGV[1] then return 0 end
442: local now = redis.call("TIME")
443: local ttl = math.max(1, math.ceil((tonumber(ARGV[3]) - tonumber(now[1])) * 1000 - tonumber(now[2]) / 1000) + 1)
444: if ttl < 9007199254740000 then
445:     redis.call("SET", KEYS[2], ARGV[2], "PX", string.format("%.0f", ttl))
446: else
447:     redis.call("SET", KEYS[2], ARGV[2])
448: end
449: return 1
450: LUA;
451:         $result = $this->client()->eval($script, [...$this->keys($key), $token, $json, sprintf("%.6F", $expires)], 2);
452: 
453:         if ($result !== 1) {
454:             throw new \RuntimeException("The Redis throttling history could not be saved under its owner lock.");
455:         }
456:     } // saveRecords
