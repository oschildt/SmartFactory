71:     /**
72:      * Validates and stores logging settings, then resets this instance's timer.
73:      * Failed initialization preserves the previous settings and timer.
74:      *
75:      * @param array{log_path: string, source_root?: string|null, write_source_file_and_line_by_debug?: mixed} $parameters
76:      * The existing writable log directory, optional existing source directory, and
77:      * optional source details flag interpreted using PHP's empty-value semantics.
78:      * An omitted or null source root defaults to the working directory at initialization.
79:      *
80:      * @return void
81:      * Completes when all settings have been accepted.
82:      *
83:      * @throws \InvalidArgumentException
84:      * If a directory setting is missing, empty, not a string, or contains a null byte.
85:      *
86:      * @throws \RuntimeException
87:      * If the log directory does not exist or is not writable, or the source root
88:      * cannot be resolved to an existing directory.
89:      *
90:      * @throws \Random\RandomException
91:      * If generating the request identifier fails because the system random source fails.
92:      *
93:      * @author Oleg Schildt
94:      */
95:     public function init(array $parameters): void
96:     {
97:         $log_path = $parameters["log_path"] ?? null;
98:         $source_root = $parameters["source_root"] ?? getcwd();
99: 
100:         if (!is_string($log_path) || $log_path === "" || str_contains($log_path, "\0")) {
101:             throw new \InvalidArgumentException("Log path must be a nonempty directory path without null bytes.");
102:         }
103: 
104:         if (!is_string($source_root) || $source_root === "" || str_contains($source_root, "\0")) {
105:             throw new \InvalidArgumentException("Source root must be a nonempty directory path without null bytes.");
106:         }
107: 
108:         $resolved_log_path = realpath($log_path);
109:         $resolved_source_root = realpath($source_root);
110: 
111:         if ($resolved_log_path === false || !is_dir($resolved_log_path) || !is_writable($resolved_log_path)) {
112:             throw new \RuntimeException(sprintf("The log directory '%s' does not exist or is not writable!", $log_path));
113:         }
114: 
115:         if ($resolved_source_root === false || !is_dir($resolved_source_root)) {
116:             throw new \RuntimeException(sprintf("The source root '%s' is not an existing directory!", $source_root));
117:         }
118: 
119:         self::$request_id ??= bin2hex(random_bytes(8));
120:         $this->log_path = rtrim(str_replace("\\", "/", $resolved_log_path), "/") . "/";
121:         $this->source_root = rtrim(str_replace("\\", "/", $resolved_source_root), "/") . "/";
122:         $this->write_source_file_and_line_by_debug = !empty($parameters["write_source_file_and_line_by_debug"]);
123:         $this->profile_time = null;
124:     } // init

142:     /**
143:      * Formats backtrace frames with a source file, including methods and closures.
144:      *
145:      * @param list<array{file?: string, line?: int, function?: string, class?: string, type?: string}> $btrace
146:      * Backtrace frames; arguments and objects are not needed.
147:      *
148:      * @return string
149:      * Returns the formatted stack, or an empty string if no frame has a source file.
150:      *
151:      * @author Oleg Schildt
152:      */
153:     protected function extract_call_stack(array $btrace): string
154:     {
155:         $lines = [];
156:         $indent = "";
157: 
158:         foreach ($btrace as $btrace_entry) {
159:             if (empty($btrace_entry["file"])) {
160:                 continue;
161:             }
162: 
163:             $function = $btrace_entry["function"] ?? "";
164: 
165:             if (preg_match("/{closure:([^}{]+)}/", $function, $matches)) {
166:                 $function = "{closure:" . $this->trim_path($matches[1]) . "}";
167:             }
168: 
169:             $call = ($btrace_entry["class"] ?? "") . ($btrace_entry["type"] ?? "") . $function;
170:             $call = $call === "" ? "" : $call . "()";
171:             $lines[] = $indent . $call . " [" . $this->trim_path($btrace_entry["file"]) . ", " .
172:                 ($btrace_entry["line"] ?? "line number undefined") . "]";
173:             $indent .= "  ";
174:         }
175: 
176:         return implode(PHP_EOL, $lines);
177:     } // extract_call_stack

179:     /**
180:      * Appends a timestamped message with a request identifier to the chosen log.
181:      * Source details and the call stack are collected only when requested.
182:      *
183:      * @param string $message
184:      * Message to log; surrounding whitespace is removed.
185:      *
186:      * @param bool $write_call_stack
187:      * Whether to include the full call stack without arguments or object contents.
188:      *
189:      * @param string $file_name
190:      * A plain file name within the configured log directory, defaulting to debug.log.
191:      * Separators, colons, control characters and trailing dots or spaces are rejected.
192:      * Existing symbolic links and nonregular files are also rejected.
193:      *
194:      * @return void
195:      * Completes after the entire entry has been appended under an exclusive lock.
196:      *
197:      * @throws \LogicException
198:      * If init() has not completed successfully.
199:      *
200:      * @throws \InvalidArgumentException
201:      * If the file name is empty, a special directory name, or contains forbidden characters.
202:      *
203:      * @throws \RuntimeException
204:      * If the target is a symbolic link or nonregular file, or the complete entry cannot be written.
205:      *
206:      * @author Oleg Schildt
207:      */
208:     public function debugMessage(string $message, bool $write_call_stack = false, string $file_name = "debug.log"): void
209:     {
210:         $logfile = $this->getLogFilePath($file_name);
211:         $prefix = "#Time: " . date("Y-m-d H:i:s") . PHP_EOL . "#Request: " . self::$request_id;
212: 
213:         if ($this->write_source_file_and_line_by_debug || $write_call_stack) {
214:             $backfiles = debug_backtrace(DEBUG_BACKTRACE_IGNORE_ARGS, $write_call_stack ? 0 : 16);
215: 
216:             if ($this->write_source_file_and_line_by_debug) {
217:                 foreach ($backfiles as $entry) {
218:                     $file = str_replace("\\", "/", $entry["file"] ?? "");
219: 
220:                     // Internal profile methods and the convenience wrapper are not the caller.
221:                     if (
222:                         $file === "" || $file === str_replace("\\", "/", __FILE__) ||
223:                         $file === str_replace("\\", "/", dirname(__DIR__) . "/short_functions_inc.php")
224:                     ) {
225:                         continue;
226:                     }
227: 
228:                     if (isset($entry["line"])) {
229:                         $prefix .= PHP_EOL . "#URI: " . ($_SERVER["REQUEST_URI"] ?? "") .
230:                             PHP_EOL . "#Source: " . $this->trim_path($file) . ", " . $entry["line"];
231:                     }
232: 
233:                     break;
234:                 }
235:             }
236: 
237:             if ($write_call_stack) {
238:                 $prefix .= PHP_EOL . "#Callstack:" . PHP_EOL . $this->extract_call_stack($backfiles);
239:             }
240:         }
241: 
242:         $message = $prefix . PHP_EOL . PHP_EOL . trim($message) . PHP_EOL . "-------" . PHP_EOL;
243:         $written = @file_put_contents($logfile, $message, FILE_APPEND | LOCK_EX);
244: 
245:         if ($written === false || $written !== strlen($message)) {
246:             throw new \RuntimeException(sprintf("Could not write the complete log entry to '%s'!", $logfile));
247:         }
248:     } // debugMessage

412:      * A nonempty file name without separators, colons, control characters or trailing dots or spaces.
413:      *
414:      * @return string
415:      * Returns the path within the configured directory; no extension is required.
416:      *
417:      * @throws \LogicException
418:      * If init() has not completed successfully.
419:      *
420:      * @throws \InvalidArgumentException
421:      * If the file name is empty, a special directory name, or contains forbidden characters.
422:      *

425:      *

428:     protected function getLogFilePath(string $file_name): string
429:     {
430:         $dir = $this->requireLogPath();
431: 
432:         if (

450:     /**
451:      * Removes the configured source root only from the beginning of a source path.
452:      * Paths outside that root are preserved, with backslashes normalized to slashes.
453:      *
454:      * @param string $path
455:      * Source path to shorten.
456:      *
457:      * @return string
458:      * Returns a relative path within source_root, or the normalized original path.
459:      *

462:     protected function trim_path(string $path): string
463:     {
464:         $path = str_replace("\\", "/", $path);
465:         $length = strlen($this->source_root);
466:         $matches = DIRECTORY_SEPARATOR === "\\"
467:             ? strncasecmp($path, $this->source_root, $length) === 0
