68:      * @param callable(string, array<array-key, mixed>): mixed $handler
69:      * A function, closure, object method, static method or invokable object accepting the

72:      * ```php

259:     /**
260:      * Fires an event synchronously, invoking handlers in registration order.
261:      *
262:      * Subscription changes take effect immediately during dispatch. Removed handlers
263:      * are skipped, newly registered handlers are appended, and suspension stops further
264:      * invocations while it remains in effect. Removing and registering a handler again
265:      * creates a new registration at the end, even if it has already run in this dispatch.
266:      * Recursive dispatches maintain independent invocation counts.
267:      *
268:      * Any exception or error from a handler stops dispatch immediately and is propagated
269:      * unchanged to the caller. The remaining handlers are not invoked.
270:      *
271:      * @param string $event
272:      * Nonempty event code. The string "0" is valid.
273:      *
274:      * @param array<array-key, mixed> $parameters
275:      * Event data passed to each handler as its second argument.
276:      *
277:      * @return int
278:      * Returns the number of handlers invoked by this dispatch, or zero if none ran.
279:      *
280:      * @throws \InvalidArgumentException
281:      * If the event code is an empty string.
282:      *
283:      * @throws \Throwable
284:      * If a handler throws an exception or error; dispatch stops and the same throwable is propagated.
285:      *
286:      * @author Oleg Schildt
287:      */
288:     public function fireEvent(string $event, array $parameters): int
289:     {
290:         if ($event === '') {
291:             throw new \InvalidArgumentException("Event is not specified!");
292:         }
293: 
294:         $cnt = 0;
295:         $last_id = 0;
296:         $pending = $this->event_table[$event] ?? [];
297: 
298:         while ($pending !== []) {
299:             foreach ($pending as $key => $registration) {
300:                 if (isset($this->suspended_events[$event])) {
301:                     return $cnt;
302:                 }
303: 
304:                 // A removed and re-added callable is a new registration, not this pending one.
305:                 if (($this->event_table[$event][$key]['id'] ?? null) !== $registration['id']) {
306:                     continue;
307:                 }
308: 
309:                 $last_id = $registration['id'];
310: 
311:                 ($registration['handler'])($event, $parameters);
312: 
313:                 $cnt++;
314:             }
315: 
316:             // Pick up additions in registration order without invoking earlier registrations twice.
317:             $pending = array_filter(
318:                 $this->event_table[$event] ?? [],
319:                 static fn(array $registration): bool => $registration['id'] > $last_id
320:             );
321:         }
322: 
323:         return $cnt;
324:     } // fireEvent

326:     /**
327:      * Builds an identity key without conflating distinct closure or target objects.
328:      *
329:      * Object references retained by registered closures prevent object ID reuse while
330:      * their registrations exist. Reflection resolves declared function/method names and
331:      * class aliases while preserving the called class of static methods and the spelling
332:      * of dynamic method names passed to __call() or __callStatic().
333:      * An explicit [$closure, '__invoke'] callable has the same identity as that closure.
334:      *
335:      * @param callable(string, array<array-key, mixed>): mixed $handler
336:      * The callable whose registration identity is required.
337:      *
338:      * @return string
339:      * Returns a key identifying a closure object, object method, static method or function.
340:      *
341:      * @author Oleg Schildt
342:      */
343:     protected function getHandlerKey(callable $handler): string
344:     {
345:         if (is_array($handler) && $handler[0] instanceof \Closure && strcasecmp($handler[1], '__invoke') === 0) {
346:             $handler = $handler[0];
347:         }
348: 
349:         if ($handler instanceof \Closure) {
350:             return 'closure:' . spl_object_id($handler);
351:         }
352: 
353:         $reflection = new \ReflectionFunction(\Closure::fromCallable($handler));
354:         $object = $reflection->getClosureThis();
355:         $class = $reflection->getClosureCalledClass();
356:         $name = $reflection->getName();
357: 
358:         if ($object !== null) {
359:             return 'method:' . spl_object_id($object) . ':' . $name;
360:         }
361: 
362:         if ($class !== null) {
363:             return 'static:' . strtolower($class->getName()) . '::' . $name;
364:         }
365: 
366:         return 'function:' . $name;
367:     } // getHandlerKey
