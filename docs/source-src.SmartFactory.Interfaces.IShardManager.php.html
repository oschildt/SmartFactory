12: /**
13:  * Registers database shards and selects their workers directly or through named groups.
14:  *
15:  * Implementations may use deterministic client affinity for random selections within each group.
16:  *
17:  * @author Oleg Schildt
18:  */
19: interface IShardManager extends IInitable
20: {
21:     /**
22:      * Initializes the manager's load-balancing and client-affinity settings.
23:      *
24:      * @param array<string, mixed> $parameters
25:      * Implementation-specific settings. The standard manager accepts the boolean
26:      * pin_shard_to_user_session (default false) and an optional zero-argument callable
27:      * user_identificator_getter returning a string, int or null. A nonempty string or
28:      * integer supplies the identity; null or an empty string uses the IP/User-Agent fallback.
29:      *
30:      * @return void
31:      * Applies routing settings while preserving registered shards and cached workers.
32:      *
33:      * @throws \Exception
34:      * If a supplied routing setting is invalid for the implementation.
35:      *
36:      * @author Oleg Schildt
37:      */
38:     public function init(array $parameters): void;
39: 
40:     /**
41:      * Registers a uniquely named shard and optionally adds it to a load-balancing group.
42:      *
43:      * @param string $shard_name
44:      * The nonempty unique shard name, including the valid name "0".
45:      *
46:      * @param array<string, mixed> $parameters
47:      * The connection parameters to the shard as an associative array in the form key => value:
48:      *
49:      * - $parameters["db_type"] - database driver (MySQL, MSSQL, PostgreSQL or Oracle)
50:      *
51:      * - $parameters["db_server"] - server address
52:      *
53:      * - $parameters["db_name"] - database name
54:      *
55:      * - $parameters["db_user"] - user name
56:      *
57:      * - $parameters["db_password"] - user password
58:      *
59:      * - $parameters["autoconnect"] - true if the connection should be established automatically
60:      *                 upon creation.
61:      *
62:      * - $parameters["read_only"] - this parameter sets the connection to the read only mode.
63:      *
64:      * @param string $load_balancing_group
65:      * The optional group name used by randomDBShard().
66:      *
67:      * @return bool
68:      * Returns true on successful registration. Invalid registrations cause an exception.
69:      *
70:      * @throws \Exception
71:      * If the shard name is empty or already registered.
72:      *
73:      * @author Oleg Schildt
74:      */
75:     public function registerShard(string $shard_name, array $parameters, string $load_balancing_group = ""): bool;
76: 
77:     /**
78:      * Returns the independently cached worker for a registered shard.
79:      *
80:      * The first access creates a worker using the shard's registered connection settings.
81:      * Later accesses through the same manager reuse it, independently of other shards' workers.
82:      *
83:      * @param string $shard_name
84:      * The name of the shard.
85:      *
86:      * @return ?\SmartFactory\DatabaseWorkers\DBWorker
87:      * Returns the cached or newly created worker, or null if the worker factory returns null.
88:      *
89:      * @throws \Exception
90:      * If the shard is unknown, the driver cannot be created, or connection initialization fails.
91:      *
92:      * @author Oleg Schildt
93:      */
94:     public function dbshard(string $shard_name): ?\SmartFactory\DatabaseWorkers\DBWorker;
95: 
96:     /**
97:      * Returns the worker of a randomly selected group member, honoring client affinity if enabled.
98:      *
99:      * @param string $load_balancing_group
100:      * The name of a nonempty registered load-balancing group.
101:      *
102:      * @return ?\SmartFactory\DatabaseWorkers\DBWorker
103:      * Returns the selected shard's worker, or null if the worker factory returns null.
104:      *
105:      * @throws \Exception
106:      * If the group is unknown, the selected shard cannot be initialized, or required
107:      * client identity is unavailable or invalid.
108:      *
109:      * @throws \Throwable
110:      * If the configured identifier getter throws an exception or error, it is propagated unchanged.
111:      *
112:      * @author Oleg Schildt
113:      */
114:     public function randomDBShard(string $load_balancing_group): ?\SmartFactory\DatabaseWorkers\DBWorker;
115: } // IShardManager
