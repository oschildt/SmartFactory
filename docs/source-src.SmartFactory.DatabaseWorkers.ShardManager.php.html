63:     /**
64:      * Initializes deterministic client affinity and the optional user-identifier getter.
65:      *
66:      * Registered shards and cached workers are preserved when initialization is repeated.
67:      * The getter is called without arguments only when affinity-enabled randomDBShard()
68:      * is invoked, so authentication may be resolved after object creation. Its nonempty
69:      * string or integer result replaces the IP/User-Agent identity. Null or an empty string
70:      * falls back to that fingerprint. No file, PHP session or shared cache is required.
71:      *
72:      * @param array{pin_shard_to_user_session?: bool, user_identificator_getter?: (callable(): (string|int|null))|null} $parameters
73:      * Optional settings. pin_shard_to_user_session defaults to false and the getter defaults
74:      * to null on every initialization. Any callable representation is accepted for the getter.
75:      *
76:      * @return void
77:      * Updates affinity settings without invoking the getter or opening database connections.
78:      *
79:      * @throws \InvalidArgumentException
80:      * If user_identificator_getter is supplied and is neither callable nor null.
81:      *
82:      * @author Oleg Schildt
83:      */

143:      * Returns the worker cached for a registered shard.
144:      *
145:      * A separate non-singleton factory request is used for each shard, then cached within
146:      * this manager. This avoids reusing another shard's connection merely because its driver class matches.
147:      *
148:      * @param string $shard_name
149:      * The registered shard name.
150:      *
151:      * @return \SmartFactory\DatabaseWorkers\DBWorker|null
152:      * Returns the cached worker, creating it from the registered connection settings on first access,
153:      * or null if the worker factory returns null.
154:      *

156:      * If the shard is unknown, the driver cannot be created, or connection initialization fails.

160:     public function dbshard(string $shard_name): ?\SmartFactory\DatabaseWorkers\DBWorker
161:     {
162:         if (!isset($this->shard_table[$shard_name])) {
163:             throw new \Exception("The shard '$shard_name' was not found!");
164:         }
165: 
166:         // Create a separate factory instance for each shard, even when several
167:         // shards use the same driver class. Cache it in this manager so later
168:         // lookups reuse the connection to that shard.
169:         if (empty($this->shard_table[$shard_name]["dbworker"])) {
170:             $this->shard_table[$shard_name]["dbworker"] = dbworker($this->shard_table[$shard_name]["parameters"], false /* not singleton */);
171:         }
172: 
173:         return $this->shard_table[$shard_name]["dbworker"];

176:     /**
177:      * Returns a random group member, or a deterministic member when client affinity is enabled.
178:      *
179:      * Affinity uses rendezvous hashing of the client identity, group name and shard name.
180:      * It requires no stored assignments and is independent of registration order. Adding
181:      * a shard only moves clients for which the new shard wins; removing one only moves
182:      * its clients. Failed worker initialization propagates without automatic failover.
183:      *
184:      * @param string $load_balancing_group
185:      * The name of a nonempty registered group.
186:      *
187:      * @return \SmartFactory\DatabaseWorkers\DBWorker|null
188:      * Returns the selected shard's cached or newly initialized worker, or null if the factory returns null.
189:      *
190:      * @throws \Exception
191:      * If the group is unknown or the selected shard cannot be initialized.
192:      *
193:      * @throws \RuntimeException
194:      * If affinity needs the fallback fingerprint but the client IP or User-Agent is missing or empty.
195:      *
196:      * @throws \UnexpectedValueException
197:      * If the configured identifier getter returns a value other than string, int or null.
198:      *
199:      * @throws \Throwable
200:      * If the identifier getter throws an exception or error, it is propagated unchanged.
201:      *
202:      * @see ShardManager::registerShard()
203:      * @see ShardManager::dbshard()
204:      *
205:      * @author Oleg Schildt
206:      */
207:     public function randomDBShard(string $load_balancing_group): ?\SmartFactory\DatabaseWorkers\DBWorker
208:     {
209:         if (empty($this->load_balancing_groups[$load_balancing_group])) {
210:             throw new \Exception("The group '$load_balancing_group' was not found!");
211:         }
212: 
213:         $shards = $this->load_balancing_groups[$load_balancing_group];
214: 
215:         if (!$this->pin_shard_to_user_session) {
216:             return $this->dbshard($shards[rand(0, count($shards) - 1)]);
217:         }
218: 
219:         $user_hash = $this->userSessionHash();
220:         $highest_score = "";
221:         $selected_shard = null;
222: 
223:         foreach ($shards as $shard_name) {
224:             $score = hash("sha256", serialize([$load_balancing_group, $user_hash, $shard_name]));
225: 
226:             // Compare fixed-length hexadecimal scores as strings to avoid numeric conversion.
227:             // A stable name-based tie-breaker also makes hash collisions independent of order.
228:             if (
229:                 $selected_shard === null
230:                 || strcmp($score, $highest_score) > 0
231:                 || ($score === $highest_score && strcmp($shard_name, $selected_shard) < 0)
232:             ) {
233:                 $highest_score = $score;
234:                 $selected_shard = $shard_name;
235:             }
236:         }
237: 
238:         return $this->dbshard($selected_shard);
239:     } // randomDBShard

241:     /**
242:      * Computes the current client's identity hash for deterministic shard selection.
243:      *
244:      * The optional zero-argument getter is evaluated on every affinity-enabled selection.
245:      * A nonempty string or any integer, including zero, replaces the IP/User-Agent identity.
246:      * Integer and equivalent string identifiers are normalized to the same identity.
247:      * A null or empty-string result falls back to REMOTE_ADDR and HTTP_USER_AGENT, as does
248:      * an omitted getter. Identity sources are separated so an identifier cannot collide
249:      * with the raw fingerprint merely because its contents match client attributes.
250:      * Forwarded headers are not interpreted; trusted proxies must normalize REMOTE_ADDR.
251:      *
252:      * @return string
253:      * Returns the SHA-256 hash of the normalized identifier or the fallback fingerprint.
254:      *
255:      * @throws \RuntimeException
256:      * If fallback identity is needed but the client IP or User-Agent is missing or empty.
257:      *
258:      * @throws \UnexpectedValueException
259:      * If the getter returns a value other than string, int or null.
260:      *
261:      * @throws \Throwable
262:      * If the getter throws an exception or error, it is propagated unchanged.
263:      *
264:      * @author Oleg Schildt
265:      */
266:     protected function userSessionHash(): string
267:     {
268:         if ($this->user_identificator_getter !== null) {
269:             $identifier = ($this->user_identificator_getter)();
270: 
271:             if ($identifier !== null && !is_string($identifier) && !is_int($identifier)) {
272:                 throw new \UnexpectedValueException("The user identifier getter must return a string, integer or null.");
273:             }
274: 
275:             if ($identifier !== null && $identifier !== "") {
276:                 return hash("sha256", serialize(["identifier", (string) $identifier]));
277:             }
278:         }
279: 
280:         $ip = $_SERVER["REMOTE_ADDR"] ?? null;
281:         $user_agent = $_SERVER["HTTP_USER_AGENT"] ?? null;
282: 
283:         if (!is_string($ip) || $ip === "" || !is_string($user_agent) || $user_agent === "") {
284:             throw new \RuntimeException("Shard affinity requires a nonempty client IP address and User-Agent.");
285:         }
286: 
287:         return hash("sha256", serialize(["fingerprint", $ip, $user_agent]));
288:     } // userSessionHash
