129:     /**
130:      * Acquires the sidecar lock shared by configuration readers and writers.
131:      *
132:      * @param int $operation
133:      * LOCK_SH for reading or LOCK_EX for merging and replacing the file.
134:      *
135:      * @return resource
136:      * Returns an open locked stream which the caller must close.
137:      *
138:      * @throws \RuntimeException
139:      * If the sidecar file cannot be opened or the requested lock cannot be acquired.
140:      *
141:      * @throws \LogicException
142:      * If the manager has not been initialized.
143:      *
144:      * @author Oleg Schildt
145:      */
146:     protected function openLock(int $operation)
147:     {
148:         $this->validateParameters();

150:         $lock = @fopen($this->save_path . ".lock", "c");

154:         }

179:     /**
180:      * Reads and decodes the file while the caller holds the sidecar lock.
181:      *
182:      * @param bool $use_cache
183:      * Whether a decoded snapshot may be reused after matching the exact payload digest.
184:      * Saves bypass the cache to merge against the persisted source.
185:      *
186:      * @return array<string, mixed>
187:      * Returns decoded values, or an empty array for an optional missing file.
188:      *
189:      * @throws \RuntimeException
190:      * If a required file is missing or the payload cannot be read.
191:      *
192:      * @throws \UnexpectedValueException
193:      * If the decoded JSON is not an object or array.
194:      *
195:      * @throws \Throwable
196:      * If decryption fails or the payload is malformed JSON.
197:      *
198:      * @author Oleg Schildt
199:      */
200:     protected function readFile(bool $use_cache): array
201:     {
202:         clearstatcache(true, $this->save_path);
203: 
204:         if (!file_exists($this->save_path) && !$this->config_file_must_exist) {
205:             return [];
206:         }
207: 
208:         $stream = @fopen($this->save_path, "rb");
209: 
210:         if ($stream === false) {
211:             throw new \RuntimeException("Cannot read the configuration file: " . $this->save_path);
212:         }
213: 
214:         try {
215:             $payload = stream_get_contents($stream);
216:         } finally {
217:             fclose($stream);
218:         }
219: 
220:         if ($payload === false) {
221:             throw new \RuntimeException("Reading the configuration file failed: " . $this->save_path);
222:         }
223: 
224:         $digest = hash('sha256', $payload);
225: 
226:         if ($use_cache && $this->use_apcu) {
227:             $cached = apcu_fetch($this->cacheKey(), $success);
228: 
229:             if ($success && is_array($cached) && ($cached['digest'] ?? null) === $digest) {
230:                 return $cached['settings'];
231:             }
232:         }
233: 
234:         $json = $this->save_encrypted ? aes_256_decrypt($payload, $this->salt_key) : $payload;
235:         $data = json_decode($json, true, 512, JSON_THROW_ON_ERROR);
236: 
237:         if (!is_array($data)) {
238:             throw new \UnexpectedValueException("Configuration JSON must contain an object or array.");
239:         }
240: 
241:         if ($use_cache && $this->use_apcu) {
242:             apcu_store($this->cacheKey(), ['digest' => $digest, 'settings' => $data]);
243:         }
244: 
245:         return $data;
246:     } // readFile

275:     /**
276:      * Merges edited parameters under an exclusive lock and atomically replaces the file.
277:      *
278:      * @param array<string, mixed> &$data
279:      * The local snapshot, replaced with the merged snapshot after a successful write.
280:      *
281:      * @return void
282:      * Preserves concurrent edits to other parameters and invalidates the decoded cache.
283:      *
284:      * @throws \RuntimeException
285:      * If the file or directory is not writable, creating, writing or flushing the temporary
286:      * file fails, permissions cannot be preserved, or replacing the target fails.
287:      *
288:      * @throws \Throwable
289:      * If initialization, locking, reading, JSON encoding or encryption fails.
290:      *
291:      * @see ConfigSettingsManager::loadJSON()
292:      *
293:      * @author Oleg Schildt
294:      */
295:     protected function saveJSON(array &$data): void
296:     {
297:         $lock = $this->openLock(LOCK_EX);
298:         $temporary = null;
299:         $stream = null;
300: 
301:         try {
302:             $merged = $this->readFile(false);
303: 
304:             foreach ($this->changed as $name => $_) {
305:                 $merged[$name] = $data[$name];
306:             }
307: 
308:             $json = json_encode($merged, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE
309:                 | JSON_PRESERVE_ZERO_FRACTION | JSON_THROW_ON_ERROR);
310:             $payload = $this->save_encrypted ? aes_256_encrypt($json, $this->salt_key) : $json;
311: 
312:             if (!is_writable(dirname($this->save_path))
313:                 || (file_exists($this->save_path) && !is_writable($this->save_path))) {
314:                 throw new \RuntimeException("The configuration file or its directory is not writable: " . $this->save_path);
315:             }
316: 
317:             $temporary = @tempnam(dirname($this->save_path), ".settings-");
318: 
319:             if ($temporary === false) {
320:                 throw new \RuntimeException("Cannot create a temporary configuration file.");
321:             }
322: 
323:             $stream = @fopen($temporary, "wb");
324: 
325:             if ($stream === false) {
326:                 throw new \RuntimeException("Cannot open the temporary configuration file.");
327:             }
328: 
329:             $length = strlen($payload);
330:             $offset = 0;
331: 
332:             while ($offset < $length) {
333:                 $written = fwrite($stream, substr($payload, $offset));
334: 
335:                 if ($written === false || $written === 0) {
336:                     throw new \RuntimeException("Writing the temporary configuration file failed.");
337:                 }
338: 
339:                 $offset += $written;
340:             }
341: 
342:             if (!fflush($stream)) {
343:                 throw new \RuntimeException("Flushing the temporary configuration file failed.");
344:             }
345: 
346:             fclose($stream);
347: 
348:             $stream = null;
349: 
350:             if (file_exists($this->save_path) && !chmod($temporary, fileperms($this->save_path) & 0777)) {
351:                 throw new \RuntimeException("Cannot preserve configuration file permissions.");
352:             }
353: 
354:             if (!@rename($temporary, $this->save_path)) {
355:                 throw new \RuntimeException("Cannot replace the configuration file: " . $this->save_path);
356:             }
357: 
358:             $temporary = null;
359:             $data = $merged;
360: 
361:             if ($this->use_apcu) {
362:                 apcu_delete($this->cacheKey());
363:             }
364:         } finally {
365:             if (is_resource($stream)) {
366:                 fclose($stream);
367:             }
368: 
369:             if (is_string($temporary) && file_exists($temporary)) {
370:                 unlink($temporary);
371:             }
372: 
373:             fclose($lock);
374:         }
375:     } // saveJSON
